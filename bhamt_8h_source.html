298static inline bhamt_hash_t
299bhamt_hash(const void* key, size_t size) {
300 const unsigned char* bytes = key;
301 bhamt_hash_t h = 0x100;
302 for (size_t i = 0; i < size; ++i) {
303 h ^= bytes[i];
304 h *= UINT64_C(1111111111111111111);
305 }
306 return h;
307}

457void*
458bhamt__do_find(
459 const bhamt_base_t* base,
460 const void* root,
461 const void* key,
462 const bhamt_spec_t* spec
463) {
464 bhamt_hash_fn_t hash_fn = bhamt__hash_fn(base);
465 bhamt_eq_fn_t eq_fn = bhamt__eq_fn(base);
466
467 const void* node = root;
468 for (bhamt_hash_t h = hash_fn(key, spec->key_size); node != NULL; h <<= BHAMT_NUM_BITS) {
469 if (eq_fn(key, (const char*)node + spec->key_offset, spec->key_size)) {
470 return (void*)node;
471 }
472 node = ((void* const*)node)[h >> (64 - BHAMT_NUM_BITS)];
473 }
474 return NULL;
475}

477void*
478bhamt__do_upsert(
479 const bhamt_base_t* base,
480 void** root,
481 const void* key,
482 const bhamt_spec_t* spec,
483 void* memctx,
484 bool* out_is_new
485) {
486 bhamt_hash_fn_t hash_fn = bhamt__hash_fn(base);
487 bhamt_eq_fn_t eq_fn = bhamt__eq_fn(base);
488
489 void** slot = root;
490 for (bhamt_hash_t h = hash_fn(key, spec->key_size); *slot != NULL; h <<= BHAMT_NUM_BITS) {
491 void* node = *slot;
492 if (eq_fn(key, (char*)node + spec->key_offset, spec->key_size)) {
493 if (out_is_new != NULL) { *out_is_new = false; }
494 return node;
495 }
496 slot = (void**)node + (h >> (64 - BHAMT_NUM_BITS));
497 }
498
499 void* new_node = BHAMT_ALLOC(spec->node_size, spec->node_align, memctx);
500 memset(new_node, 0, spec->node_size);
501 memcpy((char*)new_node + spec->key_offset, key, spec->key_size);
502 *slot = new_node;
503 if (out_is_new != NULL) { *out_is_new = true; }
504 return new_node;
505}

520void*
521bhamt__iter_next(bhamt_iter_t* iter) {
522 if (iter->len == 0) { return NULL; }
523
524 void* node = iter->stack[--iter->len];
525 void* const* children = node;
526 for (int i = 0; i < BHAMT_NUM_CHILDREN; ++i) {
527 if (children[i] != NULL) {
528 BHAMT_ASSERT(
529 iter->len < BHAMT_ITER_STACK_SIZE,
530 "%s: Iterator stack overflow (%d)",
531 (int)BHAMT_ITER_STACK_SIZE
532 );
533 iter->stack[iter->len++] = children[i];
534 }
535 }
536 return node;
537}

539void
540bhamt__do_validate(const bhamt_base_t* base, const void* root, const bhamt_spec_t* spec) {
541 bhamt_iter_t iter = { .len = 0 };
542 if (root != NULL) { iter.stack[iter.len++] = (void*)root; }
543
544 // Every node must be findable through its own key.
545 // This verifies both its position in the trie and the uniqueness of
546 // its key.
547 const void* node;
548 while ((node = bhamt__iter_next(&iter)) != NULL) {
549 const void* found = bhamt__do_find(
550 base, root, (const char*)node + spec->key_offset, spec
551 );
552 BHAMT_ASSERT(
553 found == node,
554 "%s: Node %p is not reachable through its key",
555 node
556 );
557 }
558}
