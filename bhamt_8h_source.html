303static inline bhamt_hash_t
304bhamt_hash(const void* key, size_t size) {
305 const unsigned char* bytes = key;
306 bhamt_hash_t h = 0x100;
307 for (size_t i = 0; i < size; ++i) {
308 h ^= bytes[i];
309 h *= UINT64_C(1111111111111111111);
310 }
311 return h;
312}

453void*
454bhamt__do_find(
455 const bhamt_base_t* base,
456 const void* root,
457 const void* key,
458 const bhamt_spec_t* spec
459) {
460 bhamt_hash_fn_t hash_fn = bhamt__hash_fn(base);
461 bhamt_eq_fn_t eq_fn = bhamt__eq_fn(base);
462
463 const void* node = root;
464 for (bhamt_hash_t h = hash_fn(key, spec->key_size); node != NULL; h <<= BHAMT_NUM_BITS) {
465 if (eq_fn(key, (const char*)node + spec->key_offset, spec->key_size)) {
466 return (void*)node;
467 }
468 node = ((void* const*)node)[h >> (64 - BHAMT_NUM_BITS)];
469 }
470 return NULL;
471}

473void*
474bhamt__do_upsert(
475 const bhamt_base_t* base,
476 void** root,
477 const void* key,
478 const bhamt_spec_t* spec,
479 void* memctx,
480 bool* out_is_new
481) {
482 bhamt_hash_fn_t hash_fn = bhamt__hash_fn(base);
483 bhamt_eq_fn_t eq_fn = bhamt__eq_fn(base);
484
485 void** slot = root;
486 for (bhamt_hash_t h = hash_fn(key, spec->key_size); *slot != NULL; h <<= BHAMT_NUM_BITS) {
487 void* node = *slot;
488 if (eq_fn(key, (char*)node + spec->key_offset, spec->key_size)) {
489 if (out_is_new != NULL) { *out_is_new = false; }
490 return node;
491 }
492 slot = (void**)node + (h >> (64 - BHAMT_NUM_BITS));
493 }
494
495 void* new_node = BHAMT_ALLOC(spec->node_size, spec->node_align, memctx);
496 memset(new_node, 0, spec->node_size);
497 memcpy((char*)new_node + spec->key_offset, key, spec->key_size);
498 *slot = new_node;
499 if (out_is_new != NULL) { *out_is_new = true; }
500 return new_node;
501}

516void*
517bhamt__iter_next(bhamt_iter_t* iter) {
518 if (iter->len == 0) { return NULL; }
519
520 void* node = iter->stack[--iter->len];
521 void* const* children = node;
522 for (int i = 0; i < BHAMT_NUM_CHILDREN; ++i) {
523 if (children[i] != NULL) {
524 BHAMT_ASSERT(
525 iter->len < BHAMT_ITER_STACK_SIZE,
526 "%s: Iterator stack overflow (%d)",
527 (int)BHAMT_ITER_STACK_SIZE
528 );
529 iter->stack[iter->len++] = children[i];
530 }
531 }
532 return node;
533}

535void
536bhamt__do_validate(const bhamt_base_t* base, const void* root, const bhamt_spec_t* spec) {
537 bhamt_iter_t iter = { .len = 0 };
538 if (root != NULL) { iter.stack[iter.len++] = (void*)root; }
539
540 // Every node must be findable through its own key.
541 // This verifies both its position in the trie and the uniqueness of
542 // its key.
543 const void* node;
544 while ((node = bhamt__iter_next(&iter)) != NULL) {
545 const void* found = bhamt__do_find(
546 base, root, (const char*)node + spec->key_offset, spec
547 );
548 BHAMT_ASSERT(
549 found == node,
550 "%s: Node %p is not reachable through its key",
551 node
552 );
553 }
554}
