392bresmon_mark_dirty(bresmon_watch_t* watch) {

400// A change notification only signals that a file is being written to, there is
401// none for when the writer is done and closes it.
402// Moreover, a single write can notify several times.
403// Thus, a changed file is only reported once it has been quiet for a while and
404// it can be opened while denying write access to others.
405// The latter fails for as long as any writer still has the file open.
406static inline int
407bresmon_confirm_writes(bresmon_t* mon) {
408 if (mon->num_dirty_watches == 0) { return 0; }
409
410 int num_events = 0;
411 ULONGLONG now = GetTickCount64();
412
413 for (
414 bresmon_dirmon_link_t* itr = mon->dirmons.next;
415 itr != &mon->dirmons;
416 itr = itr->next
417 ) {
418 bresmon_dirmon_t* dirmon = (bresmon_dirmon_t*)((char*)itr - offsetof(bresmon_dirmon_t, link));
419
420 for (
421 bresmon_watch_link_t* watch_itr = dirmon->watches.next;
422 watch_itr != &dirmon->watches;
423 watch_itr = watch_itr->next
424 ) {
425 bresmon_watch_t* watch = (bresmon_watch_t*)((char*)watch_itr - offsetof(bresmon_watch_t, link));
426 if (!watch->dirty) { continue; }
427 if (now - watch->last_change_time < (ULONGLONG)(BRESMON_SETTLE_TIME_MS)) { continue; }
428
429 // The handle is closed right away since it also prevents a writer
430 // from opening the file.
431 // Read access has to be requested, otherwise, the share mode is
432 // not checked.
433 HANDLE file = CreateFileA(
434 watch->full_path,
435 GENERIC_READ,
436 FILE_SHARE_READ | FILE_SHARE_DELETE,
437 NULL,
438 OPEN_EXISTING,
439 FILE_ATTRIBUTE_NORMAL,
440 NULL
441 );
442 bool changed = true;
443 if (file != INVALID_HANDLE_VALUE) {
444 CloseHandle(file);
445 } else {
446 DWORD error = GetLastError();
447 if (
448 error == ERROR_SHARING_VIOLATION
449 // Pending deletion
450 || error == ERROR_ACCESS_DENIED
451 ) {
452 // Still in use, check again later
453 continue;
454 } else if (
455 error == ERROR_FILE_NOT_FOUND
456 || error == ERROR_PATH_NOT_FOUND
457 ) {
458 // Deleted after the change
459 changed = false;
460 }
461 // Otherwise, there is no telling, conservatively treat the
462 // file as changed.
463 }
464
465 watch->dirty = false;
466 --mon->num_dirty_watches;
467 if (changed) {
468 ++watch->latest_version;
469 ++num_events;
470 }
471 }
472 }
473
474 return num_events;
475}

503void
504bresmon_destroy(bresmon_t* mon) {
505 while (mon->dirmons.next != &mon->dirmons) {
506 bresmon_dirmon_t* dirmon = (bresmon_dirmon_t*)((char*)mon->dirmons.next - offsetof(bresmon_dirmon_t, link));
507 if (dirmon->watches.next != &dirmon->watches) {
508 bresmon_watch_t* watch = (bresmon_watch_t*)((char*)dirmon->watches.next - offsetof(bresmon_watch_t, link));
509 bresmon_unwatch(watch);
510 }
511 }
512
513#if defined(__linux__)
514 close(mon->inotifyfd);
515#elif defined(_WIN32)
516 // Drain the cancelled reads so their dirmons get freed
517 while (mon->num_closing_dirmons > 0) {
518 bresmon_should_reload(mon, true);
519 }
520 CloseHandle(mon->iocp);
521#endif
522
523 bresmon_free(mon, mon->memctx);
524}

707bresmon_unwatch(bresmon_watch_t* watch) {
708 if (watch == NULL) { return; }
709
710 bresmon_t* mon = watch->dirmon->root;
711 watch->link.prev->next = watch->link.next;
712 watch->link.next->prev = watch->link.prev;
713
714#if defined(_WIN32)
715 if (watch->dirty) { --mon->num_dirty_watches; }
716#endif
717
718 bresmon_dirmon_t* dirmon = watch->dirmon;
719 if (dirmon->watches.next == &dirmon->watches) {
720 dirmon->link.next->prev = dirmon->link.prev;
721 dirmon->link.prev->next = dirmon->link.next;
722 bool free_dirmon = true;
723
724#if defined(__linux__)
725 inotify_rm_watch(mon->inotifyfd, dirmon->watchd);
726#elif defined(_WIN32)
727 CancelIo(dirmon->dir_handle);
728 CloseHandle(dirmon->dir_handle);
729 dirmon->dir_handle = INVALID_HANDLE_VALUE;
730 if (dirmon->read_pending) {
731 // The cancelled read still completes through the port with this
732 // dirmon as its key.
733 // Keep it allocated until bresmon_should_reload dequeues that
734 // completion and frees it.
735 ++mon->num_closing_dirmons;
736 free_dirmon = false;
737 }
738#endif
739
740 if (free_dirmon) {
741 bresmon_free(dirmon, mon->memctx);
742 }
743 }
744
745 bresmon_free(watch, mon->memctx);
746}

748int
749bresmon_should_reload(bresmon_t* mon, bool wait) {
750 int num_events = 0;
751
752#if defined(__linux__)
753 if (wait) {
754 struct pollfd pollfd = {
755 .fd = mon->inotifyfd,
756 .events = POLLIN,
757 };
758 poll(&pollfd, 1, -1);
759 }
760
761 _Alignas(struct inotify_event) char event_buf[sizeof(struct inotify_event) + NAME_MAX + 1];
762
763 while (true) {
764 ssize_t num_bytes_read = read(mon->inotifyfd, event_buf, sizeof(event_buf));
765
766 if (num_bytes_read <= 0) {
767 break;
768 }
769
770 for (
771 char* event_itr = event_buf;
772 event_itr - event_buf < num_bytes_read;
773 ) {
774 struct inotify_event* event = (struct inotify_event*)event_itr;
775 event_itr += sizeof(struct inotify_event) + event->len;
776
777 for (
778 bresmon_dirmon_link_t* itr = mon->dirmons.next;
779 itr != &mon->dirmons;
780 itr = itr->next
781 ) {
782 bresmon_dirmon_t* dirmon = (bresmon_dirmon_t*)((char*)itr - offsetof(bresmon_dirmon_t, link));
783
784 if (dirmon->watchd == event->wd) {
785 for (
786 bresmon_watch_link_t* watch_itr = dirmon->watches.next;
787 watch_itr != &dirmon->watches;
788 watch_itr = watch_itr->next
789 ) {
790 bresmon_watch_t* watch = (bresmon_watch_t*)((char*)watch_itr - offsetof(bresmon_watch_t, link));
791 if (strcmp(watch->filename, event->name) == 0) {
792 ++watch->latest_version;
793 ++num_events;
794 }
795 }
796
797 break;
798 }
799 }
800 }
801 }
802#elif defined(_WIN32)
803 OVERLAPPED_ENTRY overlapped_entry;
804 ULONG num_entries = 0;
805 bool dequeued_any = false;
806
807 while (true) {
808 DWORD timeout = 0;
809 if (wait) {
810 if (mon->num_dirty_watches > 0) {
811 // Completion of a write has to be polled
812 timeout = BRESMON_WIN32_POLL_INTERVAL_MS;
813 } else if (!dequeued_any) {
814 // Only wait for the initial event
815 timeout = INFINITE;
816 }
817 }
818
819 BOOL dequeued = GetQueuedCompletionStatusEx(
820 mon->iocp,
821 &overlapped_entry, 1,
822 &num_entries,
823 timeout,
824 TRUE
825 );
826
827 if (!dequeued || num_entries == 0) {
828 // No event
829 num_events += bresmon_confirm_writes(mon);
830 if (wait && num_events == 0 && mon->num_dirty_watches > 0) {
831 // Keep waiting for the pending writes
832 continue;
833 } else {
834 break;
835 }
836 }
837
838 dequeued_any = true;
839 if (overlapped_entry.lpOverlapped == NULL) {
840 // Not one of ours
841 continue;
842 }
843
844 bresmon_dirmon_t* dirmon = (bresmon_dirmon_t *)overlapped_entry.lpCompletionKey;
845 dirmon->read_pending = false;
846
847 if (dirmon->dir_handle == INVALID_HANDLE_VALUE) {
848 // The read cancelled by bresmon_unwatch has completed, the dirmon
849 // is no longer referenced by the port and can go
850 --mon->num_closing_dirmons;
851 bresmon_free(dirmon, mon->memctx);
852 continue;
853 }
854
855 if (overlapped_entry.dwNumberOfBytesTransferred == 0) {
856 // The notification buffer overflowed (or the read failed) and the
857 // changes were lost.
858 // Since there is no telling which files changed, conservatively
859 // treat every watched file in the directory as changed.
860 for (
861 bresmon_watch_link_t* watch_itr = dirmon->watches.next;
862 watch_itr != &dirmon->watches;
863 watch_itr = watch_itr->next
864 ) {
865 bresmon_watch_t* watch = (bresmon_watch_t*)((char*)watch_itr - offsetof(bresmon_watch_t, link));
866 bresmon_mark_dirty(watch);
867 }
868 } else {
