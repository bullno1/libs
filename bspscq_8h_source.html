178bool
179bspscq_produce(bspscq_t* queue, void* item, bool wait) {
180 if (atomic_load(&queue->count) == (int)queue->size) {
181 if (!wait) { return false; }
182
183 mtx_lock(&queue->can_produce.mtx);
184 while (queue->count == (int)queue->size) {
185 cnd_wait(&queue->can_produce.cnd, &queue->can_produce.mtx);
186 }
187 mtx_unlock(&queue->can_produce.mtx);
188 }
189
190 unsigned int tail = atomic_fetch_add(&queue->tail, 1);
191 queue->values[tail & (queue->size - 1)] = item;
192 if (atomic_fetch_add(&queue->count, 1) == 0) {
193 bspscq_signal_raise(&queue->can_consume);
194 }
195
196 return true;
197}

199void*
200bspscq_consume(bspscq_t* queue, bool wait) {
201 if (atomic_load(&queue->count) == 0) {
202 if (!wait) { return NULL; }
203
204 mtx_lock(&queue->can_consume.mtx);
205 while (queue->count == 0) {
206 cnd_wait(&queue->can_consume.cnd, &queue->can_consume.mtx);
207 }
208 mtx_unlock(&queue->can_consume.mtx);
209 }
210
211 unsigned int head = atomic_fetch_add(&queue->head, 1);
212 void* item = queue->values[head & (queue->size - 1)];
213 if (atomic_fetch_add(&queue->count, -1) == (int)queue->size) {
214 bspscq_signal_raise(&queue->can_produce);
215 }
216
217 return item;
218}
