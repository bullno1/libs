201#define BARENA_POISON(ptr, size) __asan_poison_memory_region((ptr), (size))

257void
258barena_pool_cleanup(barena_pool_t* pool) {
259 for (
260 barena_chunk_t* chunk_itr = pool->free_chunks;
261 chunk_itr != NULL;
262 ) {
263 barena_chunk_t* next = chunk_itr->next;
264 size_t chunk_size = (size_t)(chunk_itr->end - (char*)chunk_itr);
265 // Whatever reuses this memory next must not find our poison in it
266 BARENA_UNPOISON(chunk_itr, chunk_size);
267 barena_os_page_free(chunk_itr, chunk_size);
268 chunk_itr = next;
269 }
270
271 pool->free_chunks = NULL;
272}

286static inline void*
287barena_alloc_from_chunk(barena_chunk_t* chunk, size_t size, size_t alignment) {
288 size_t space_available = chunk != NULL ? chunk->end - chunk->begin : 0;
289 if (space_available < size) { return NULL; }
290
291 intptr_t result = barena_align_ptr((intptr_t)chunk->bump_ptr, alignment);
292 intptr_t new_bump_ptr = result + (ptrdiff_t)size + BARENA_REDZONE_SIZE;
293 if (new_bump_ptr > (intptr_t)chunk->end) { return NULL; }
294
295 chunk->bump_ptr = (char*)new_bump_ptr;
296 BARENA_UNPOISON((void*)result, size);
297 return (void*)result;
298}

300void*
301barena_memalign(barena_t* arena, size_t size, size_t alignment) {
302 if (size == 0) { return NULL; }
303 if (alignment < BARENA_MIN_ALIGNMENT) { alignment = BARENA_MIN_ALIGNMENT; }
304
305 barena_chunk_t* current_chunk = arena->current_chunk;
306 void* result = barena_alloc_from_chunk(current_chunk, size, alignment);
307 if (result != NULL) { return result; }
308
309 // New chunk needed
310 barena_pool_t* pool = arena->pool;
311 size_t chunk_size = pool->chunk_size;
312 size_t required_size = (size_t)barena_align_ptr(
313 // A redzone on each side of the block
314 (intptr_t)(sizeof(barena_chunk_t) + BARENA_REDZONE_SIZE + size + BARENA_REDZONE_SIZE),
315 pool->os_page_size
316 );
317 size_t alloc_size = chunk_size >= required_size ? chunk_size : required_size;
318
319 barena_chunk_t* new_chunk;
320 if (
321 pool->free_chunks != NULL
322 && (size_t)(pool->free_chunks->end - (char*)pool->free_chunks) >= alloc_size
323 ) {
324 new_chunk = pool->free_chunks;
325 pool->free_chunks = new_chunk->next;
326 } else {
327 new_chunk = barena_os_page_alloc(alloc_size);
328 new_chunk->end = (char*)new_chunk + alloc_size;
329 }
330
331 // The leading redzone guards the chunk header against an underrun of the
332 // first block
333 new_chunk->bump_ptr = new_chunk->begin + BARENA_REDZONE_SIZE;
334 new_chunk->next = arena->current_chunk;
335 arena->current_chunk = new_chunk;
336 BARENA_POISON(new_chunk->begin, (size_t)(new_chunk->end - new_chunk->begin));
337
338 return barena_alloc_from_chunk(new_chunk, size, alignment);
339}

348void
349barena_restore(barena_t* arena, barena_snapshot_t snapshot) {
350 barena_pool_t* pool = arena->pool;
351 barena_chunk_t* itr = arena->current_chunk;
352 while (
353 itr != NULL
354 && !(itr->begin <= snapshot && snapshot <= itr->end)
355 ) {
356 barena_chunk_t* next = itr->next;
357
358 BARENA_POISON(itr->begin, (size_t)(itr->end - itr->begin));
359 itr->next = pool->free_chunks;
360 pool->free_chunks = itr;
361
362 itr = next;
363 }
364
365 if (snapshot != NULL) {
366 BARENA_POISON(snapshot, (size_t)(itr->end - snapshot));
367 itr->bump_ptr = snapshot;
368 }
369 arena->current_chunk = itr;
370}
