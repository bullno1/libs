226static inline void*
227barena_alloc_from_chunk(barena_chunk_t* chunk, size_t size, size_t alignment) {
228 size_t space_available = chunk != NULL ? chunk->end - chunk->begin : 0;
229 if (space_available < size) { return NULL; }
230
231 intptr_t result = barena_align_ptr((intptr_t)chunk->bump_ptr, alignment);
232 intptr_t new_bump_ptr = result + (ptrdiff_t)size;
233 if (new_bump_ptr > (intptr_t)chunk->end) { return NULL; }
234
235 chunk->bump_ptr = (char*)new_bump_ptr;
236 return (void*)result;
237}

239void*
240barena_memalign(barena_t* arena, size_t size, size_t alignment) {
241 if (size == 0) { return NULL; }
242
243 barena_chunk_t* current_chunk = arena->current_chunk;
244 void* result = barena_alloc_from_chunk(current_chunk, size, alignment);
245 if (result != NULL) { return result; }
246
247 // New chunk needed
248 barena_pool_t* pool = arena->pool;
249 size_t chunk_size = pool->chunk_size;
250 size_t required_size = (size_t)barena_align_ptr(
251 (intptr_t)(sizeof(barena_chunk_t) + size),
252 pool->os_page_size
253 );
254 size_t alloc_size = chunk_size >= required_size ? chunk_size : required_size;
255
256 barena_chunk_t* new_chunk;
257 if (
258 pool->free_chunks != NULL
259 && (size_t)(pool->free_chunks->end - (char*)pool->free_chunks) >= alloc_size
260 ) {
261 new_chunk = pool->free_chunks;
262 pool->free_chunks = new_chunk->next;
263 } else {
264 new_chunk = barena_os_page_alloc(alloc_size);
265 new_chunk->end = (char*)new_chunk + alloc_size;
266 }
267
268 new_chunk->bump_ptr = new_chunk->begin;
269 new_chunk->next = arena->current_chunk;
270 arena->current_chunk = new_chunk;

272 return barena_alloc_from_chunk(new_chunk, size, alignment);

284 barena_pool_t* pool = arena->pool;
285 barena_chunk_t* itr = arena->current_chunk;
