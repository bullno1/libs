386// The only place that moves the cursor forward.
387// It maintains line tracking by counting newlines in the consumed span.
388static bscn_str_t
389bscn__advance(bscn_t* scn, bscn_index_t len) {
390 bscn_str_t view = {
391 .chars = scn->input.chars + scn->bscn__cursor,
392 .len = len,
393 };
394
395 for (bscn_index_t i = 0; i < len; ++i) {
396 if (view.chars[i] == '\n') {
397 scn->bscn__line += 1;
398 scn->bscn__line_start = scn->bscn__cursor + i + 1;
399 }
400 }
401 scn->bscn__cursor += len;
402
403 return view;
404}

541bscn_accept_until_str(bscn_t* scn, bscn_str_t str) {
542 if (str.len <= 0) { return bscn__empty(scn); }

598bscn_str_t
599bscn_accept_new_line(bscn_t* scn) {
600 const char* cur = scn->input.chars + scn->bscn__cursor;
601 bscn_index_t remaining = scn->input.len - scn->bscn__cursor;
602
603 if (remaining >= 2 && cur[0] == '\r' && cur[1] == '\n') {
604 return bscn__advance(scn, 2);
605 } else if (remaining >= 1 && cur[0] == '\n') {
606 return bscn__advance(scn, 1);
607 } else {
608 return bscn__empty(scn);
609 }

612bscn_str_t
613bscn_next_line(bscn_t* scn) {
614 const char* chars = scn->input.chars;
615 bscn_index_t begin = scn->bscn__cursor;
616
617 bscn_index_t i = begin;
618 while (i < scn->input.len && chars[i] != '\n') { ++i; }
619
620 bscn_index_t content_end = i;
621 if (content_end > begin && chars[content_end - 1] == '\r') { --content_end; }
622
623 // Consume the terminator but exclude it from the view
624 bscn_index_t consumed = i < scn->input.len ? i + 1 - begin : i - begin;
625 bscn__advance(scn, consumed);
626
627 return (bscn_str_t){ .chars = chars + begin, .len = content_end - begin };
628}
