442static inline uint64_t
443bhash__chibihash64__load32le(const uint8_t *p) {
444 return (uint64_t)p[0] << 0 | (uint64_t)p[1] << 8 |
445 (uint64_t)p[2] << 16 | (uint64_t)p[3] << 24;
446}
447
448static inline uint64_t
449bhash__chibihash64__load64le(const uint8_t *p) {
450 return bhash__chibihash64__load32le(p) | (bhash__chibihash64__load32le(p+4) << 32);
451}
452
453static inline uint64_t
454bhash__chibihash64__rotl(uint64_t x, int n) {
455 return (x << n) | (x >> (-n & 63));
456}
457
458static inline uint64_t
459bhash__chibihash64(const void *keyIn, ptrdiff_t len, uint64_t seed) {
460 const uint8_t *p = (const uint8_t *)keyIn;
461 ptrdiff_t l = len;
462
463 const uint64_t K = UINT64_C(0x2B7E151628AED2A7); // digits of e
464 uint64_t seed2 = bhash__chibihash64__rotl(seed-K, 15) + bhash__chibihash64__rotl(seed-K, 47);
465 uint64_t h[4] = { seed, seed+K, seed2, seed2+(K*K^K) };
466
467 // depending on your system unrolling might (or might not) make things
468 // a tad bit faster on large strings. on my system, it actually makes
469 // things slower.
470 // generally speaking, the cost of bigger code size is usually not
471 // worth the trade-off since larger code-size will hinder inlinability
472 // but depending on your needs, you may want to uncomment the pragma
473 // below to unroll the loop.
474 //#pragma GCC unroll 2
475 for (; l >= 32; l -= 32) {
476 for (int i = 0; i < 4; ++i, p += 8) {
477 uint64_t stripe = bhash__chibihash64__load64le(p);
478 h[i] = (stripe + h[i]) * K;
479 h[(i+1)&3] += bhash__chibihash64__rotl(stripe, 27);
480 }
481 }
482
483 for (; l >= 8; l -= 8, p += 8) {
484 h[0] ^= bhash__chibihash64__load32le(p+0); h[0] *= K;
485 h[1] ^= bhash__chibihash64__load32le(p+4); h[1] *= K;
486 }
487
488 if (l >= 4) {
489 h[2] ^= bhash__chibihash64__load32le(p);
490 h[3] ^= bhash__chibihash64__load32le(p + l - 4);
491 } else if (l > 0) {
492 h[2] ^= p[0];
493 h[3] ^= p[l/2] | ((uint64_t)p[l-1] << 8);
494 }
495
496 h[0] += bhash__chibihash64__rotl(h[2] * K, 31) ^ (h[2] >> 31);
497 h[1] += bhash__chibihash64__rotl(h[3] * K, 31) ^ (h[3] >> 31);
498 h[0] *= K; h[0] ^= h[0] >> 31;
499 h[1] += h[0];
500
501 uint64_t x = (uint64_t)len * K;
502 x ^= bhash__chibihash64__rotl(x, 29);
503 x += seed;
504 x ^= h[1];
505
506 x ^= bhash__chibihash64__rotl(x, 15) ^ bhash__chibihash64__rotl(x, 42);
507 x *= K;
508 x ^= bhash__chibihash64__rotl(x, 13) ^ bhash__chibihash64__rotl(x, 31);
509
510 return x;
511}

700static inline bhash_index_t
701bhash_lookup_index(bhash_hash_t hash, bhash_index_t exp, bhash_index_t idx) {
702 uint32_t mask = ((uint32_t)1 << exp) - 1;
703 uint32_t step = (uint32_t)((hash >> (64 - exp)) | 1);
704 return (idx + step) & mask;
705}

707static inline void
708bhash_maybe_grow(bhash_base_t* bhash) {
709 if (bhash->free_space > 0) { return; }
710
711 bhash_index_t exp = bhash->exp;
712 bhash_index_t hash_capacity = 1 << exp;
713 bhash_index_t data_capacity = hash_capacity * bhash->load_percent / 100;
714 bhash_index_t num_tombstones = data_capacity - bhash->len;
715 // Grow if there are not too many tombstone. Otherwise, do in-place rehash
716 if (num_tombstones < data_capacity * bhash->tombstone_percent / 100) {
717 bhash->exp = exp += 1;
718 hash_capacity = 1 << exp;
719 data_capacity = hash_capacity * bhash->load_percent / 100;
720 bhash->indices = BHASH_REALLOC(bhash->indices, sizeof(bhash_index_t) * hash_capacity, bhash->memctx);
721 bhash->hashes = BHASH_REALLOC(bhash->hashes, sizeof(bhash_hash_t) * data_capacity, bhash->memctx);
722 if (bhash->r_indices) {
723 bhash->r_indices = BHASH_REALLOC(bhash->r_indices, sizeof(bhash_index_t) * data_capacity, bhash->memctx);
724 }
725
726 bhash_index_t extra_space = bhash->r_indices != NULL ? 1 : 0;
727 *bhash_keys_ptr(bhash) = BHASH_REALLOC(*bhash_keys_ptr(bhash), bhash->key_size * (data_capacity + extra_space), bhash->memctx);
728 if (bhash->value_size > 0) {
729 *bhash_values_ptr(bhash) = BHASH_REALLOC(*bhash_values_ptr(bhash), bhash->value_size * (data_capacity + extra_space), bhash->memctx);
730 }
731 }
732
733 bhash_index_t len = bhash->len;
734 bhash_index_t* indices = bhash->indices;
735 bhash_index_t* r_indices = bhash->r_indices;
736 bhash_hash_t* hashes = bhash->hashes;
737 memset(bhash->indices, 0, sizeof(bhash_index_t) * hash_capacity);
738 for (bhash_index_t i = 0; i < len; ++i) {
739 bhash_hash_t hash = hashes[i];
740 for (bhash_index_t hash_index = (bhash_index_t)hash;;) {
741 hash_index = bhash_lookup_index(hash, exp, hash_index);
742 bhash_index_t data_index = indices[hash_index];
743 if (data_index == BHASH_EMPTY) {
744 indices[hash_index] = i + 1;
745 if (r_indices != NULL) { r_indices[i] = hash_index; }
746 break;
747 }
748 }
749 }
750 bhash->free_space = data_capacity - bhash->len;
751}

753static inline void
754bhash_find_impl(
755 const bhash_base_t* bhash,
756 bhash_index_t* out_data_index,
757 bhash_index_t* out_hash_index,
758 const void* key
759) {
760 bhash_hash_t hash = bhash->hash(key, bhash->key_size);
761 bhash_index_t* indices = bhash->indices;
762 bhash_index_t exp = bhash->exp;
763 bhash_hash_t* hashes = bhash->hashes;
764 for (bhash_index_t hash_index = (bhash_index_t)hash;;) {
765 hash_index = bhash_lookup_index(hash, exp, hash_index);
766 bhash_index_t data_index = indices[hash_index];
767 if (data_index == BHASH_EMPTY) {
768 *out_data_index = *out_hash_index = -1;
769 return;
770 } else if (data_index == BHASH_TOMBSTONE) {
771 continue;
772 } else if (
773 hashes[data_index - 1] == hash
774 && bhash->eq(key, bhash_key_at(bhash, data_index - 1), bhash->key_size)
775 ) {
776 *out_data_index = data_index - 1;
777 *out_hash_index = hash_index;
778 return;
779 }
780 }

802bhash__do_init(
803 bhash_base_t* bhash,
804 size_t key_size,
805 size_t value_size,
806 const bhash_config_t* opt_config
807) {
808 bhash_config_t config = bhash_merge_config(opt_config);
809
810 bhash_index_t hash_capacity = 1 << config.initial_exp;
811 bhash_index_t data_capacity = hash_capacity * config.load_percent / 100;
812 bhash_index_t extra_space = !config.put_only ? 1 : 0; // Extra temp space for swapping
813 (*bhash) = (bhash_base_t){

815 .hash = config.hash,
816 .eq = config.eq,

819 .key_size = key_size,

821 .indices = BHASH_REALLOC(NULL, sizeof(bhash_index_t) * hash_capacity, config.memctx),
822 .hashes = BHASH_REALLOC(NULL, sizeof(bhash_hash_t) * data_capacity, config.memctx),
823 .len = 0,
824 .exp = config.initial_exp,
825 .free_space = data_capacity,
826 };
827 memset(bhash->indices, 0, sizeof(bhash_index_t) * hash_capacity);
828
829 if (!config.put_only) {

831 }
832

834 if (value_size > 0) {
835 *bhash_values_ptr(bhash) = BHASH_REALLOC(NULL, value_size * (data_capacity + extra_space), config.memctx);
836 }
837}

851bhash_alloc_result_t
852bhash__do_alloc(bhash_base_t* bhash, const void* key) {
853 bhash_maybe_grow(bhash);
854 bhash_hash_t hash = bhash->hash(key, bhash->key_size);
855 bhash_index_t dest_slot = -1;
856 bhash_index_t exp = bhash->exp;
857 bhash_index_t* indices = bhash->indices;
858 bhash_hash_t* hashes = bhash->hashes;
859 for (bhash_index_t hash_index = (bhash_index_t)hash;;) {
860 hash_index = bhash_lookup_index(hash, exp, hash_index);
861 bhash_index_t data_index = indices[hash_index];
862 if (data_index == BHASH_EMPTY) {
863 bhash->free_space -= (dest_slot == -1); // New empty slot allocated
864 dest_slot = dest_slot == -1 ? hash_index : dest_slot;
865 data_index = bhash->len++;
866 indices[dest_slot] = data_index + 1;
867 if (bhash->r_indices) { bhash->r_indices[data_index] = dest_slot; }
868 hashes[data_index] = hash;
869 return (bhash_alloc_result_t){
870 .index = data_index,
871 .is_new = true,
872 };
873 } else if (data_index == BHASH_TOMBSTONE) {
874 dest_slot = dest_slot == -1 ? hash_index : dest_slot;
875 } else if (
876 hashes[data_index - 1] == hash
877 && bhash->eq(key, bhash_key_at(bhash, data_index - 1), bhash->key_size)
878 ) {
879 return (bhash_alloc_result_t){
880 .index = data_index - 1,
881 .is_new = false,
882 };
883 }
884 }
885}

914bhash_index_t
915bhash__do_remove(bhash_base_t* bhash, const void* key) {
916 if (bhash->r_indices == NULL) { return -1; }
917
918 bhash_index_t remove_index, remove_r_index;
919 bhash_index_t end_index = bhash->len;
920 bhash_index_t tail_index = end_index - 1;
921 bhash_find_impl(bhash, &remove_index, &remove_r_index, key);
922 if (remove_index == -1) { return remove_index; }
923
924 // Move the last element into the deleted slot and delete the last element
925 bhash_index_t tail_r_index = bhash->r_indices[tail_index];
926 bhash->indices[tail_r_index] = remove_index + 1;
927 bhash->indices[remove_r_index] = BHASH_TOMBSTONE;
928 bhash->r_indices[remove_index] = tail_r_index;
929 bhash->hashes[remove_index] = bhash->hashes[tail_index];
930
931 // Rotate key and values then point user code to the temp position at the end
932 memcpy(bhash_key_at(bhash, end_index), bhash_key_at(bhash, remove_index), bhash->key_size);
933 memcpy(bhash_key_at(bhash, remove_index), bhash_key_at(bhash, tail_index), bhash->key_size);
934 if (bhash->value_size > 0) {
935 memcpy(bhash_value_at(bhash, end_index), bhash_value_at(bhash, remove_index), bhash->value_size);
936 memcpy(bhash_value_at(bhash, remove_index), bhash_value_at(bhash, tail_index), bhash->value_size);
937 }
938
939 bhash->len -= 1;
940 return end_index;
941}

943void
944bhash__do_validate(bhash_base_t* bhash) {
945 bhash_index_t len = bhash->len;
946 bhash_hash_t* hashes = bhash->hashes;
947 bhash_index_t* indices = bhash->indices;
948 bhash_index_t* r_indices = bhash->r_indices;
949 for (bhash_index_t i = 0; i < len; ++i) {
950 bhash_hash_t stored_hash = hashes[i];
951 bhash_hash_t computed_hash = bhash->hash(bhash_key_at(bhash, i), bhash->key_size);
952 BHASH_ASSERT(stored_hash == computed_hash, "%s: Hash mismatch at %d", i);
953 bhash_index_t r_index = r_indices[i];
954 bhash_index_t index = indices[r_index];
955 BHASH_ASSERT(index == i + 1, "%s: Index mismatch at %d", i);
956 }
957 bhash_index_t hash_capacity = (bhash_index_t)1 << bhash->exp;
958 bhash_index_t data_capacity = hash_capacity * bhash->load_percent / 100;
959 BHASH_ASSERT(len <= data_capacity, "%s: Invalid length %d (max: %d)", len, data_capacity);
960 for (bhash_index_t i = 0; i < hash_capacity; ++i) {
961 bhash_index_t index = indices[i];
962 if (index <= 0) {
963 BHASH_ASSERT(
964 index == BHASH_EMPTY || index == BHASH_TOMBSTONE,
965 "%s: Invalid negative index %d",
966 index
967 );
968 } else {
969 BHASH_ASSERT(
970 (index - 1) <= len,
971 "%s: Invalid positive index %d",
972 index
973 );
974 bhash_index_t r_index = r_indices[index - 1];
975 BHASH_ASSERT(
976 i == r_index,
977 "%s: Index mismatch at %d",
978 i
979 );
980 }
981 }
