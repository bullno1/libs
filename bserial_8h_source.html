1350 // bserial_end_op that ends its own op should pop the scope.
1351 if (
1352 (ctx->scope->type == BSERIAL_SCOPE_BLOB && op == BSERIAL_OP_BLOB)
1353 || (ctx->scope->type == BSERIAL_SCOPE_RECORD && op == BSERIAL_OP_RECORD)
1354 ) {
1355 BSERIAL_CHECK_STATUS(bserial_pop_scope(ctx));
1356 }
1357
1358 // Auto pop when enough ops are executed.
1359 // Array and table do not have an "end" function call and the number of
1360 // elements are automatically tracked through bserial_end_op.
1361 while (
1362 (ctx->scope->type == BSERIAL_SCOPE_ARRAY
1363 || ctx->scope->type == BSERIAL_SCOPE_TABLE)
1364 && ctx->scope->iterator == ctx->scope->len
1365 ) {
1366 BSERIAL_CHECK_STATUS(bserial_pop_scope(ctx));
1367 }
1368
1369 return BSERIAL_OK;
1370}

1627
1628// small, fast 64 bit hash function (version 2).
1629//
1630// https://github.com/N-R-K/ChibiHash
1631//
1632// This is free and unencumbered software released into the public domain.
1633// For more information, please refer to <https://unlicense.org/>
1634
1635static inline uint64_t
1636bserial__chibihash64__load32le(const uint8_t *p) {
1637 return (uint64_t)p[0] << 0 | (uint64_t)p[1] << 8 |
1638 (uint64_t)p[2] << 16 | (uint64_t)p[3] << 24;
1639}
1640
1641static inline uint64_t
1642bserial__chibihash64__load64le(const uint8_t *p) {
1643 return bserial__chibihash64__load32le(p) | (bserial__chibihash64__load32le(p+4) << 32);
1644}
1645
1646static inline uint64_t
1647bserial__chibihash64__rotl(uint64_t x, int n) {
1648 return (x << n) | (x >> (-n & 63));
1649}
1650
1651static inline uint64_t
1652bserial__chibihash64(const void *keyIn, ptrdiff_t len, uint64_t seed) {
1653 const uint8_t *p = (const uint8_t *)keyIn;
1654 ptrdiff_t l = len;
1655
1656 const uint64_t K = UINT64_C(0x2B7E151628AED2A7); // digits of e
1657 uint64_t seed2 = bserial__chibihash64__rotl(seed-K, 15) + bserial__chibihash64__rotl(seed-K, 47);
1658 uint64_t h[4] = { seed, seed+K, seed2, seed2+(K*K^K) };
1659
1660 // depending on your system unrolling might (or might not) make things
1661 // a tad bit faster on large strings. on my system, it actually makes
1662 // things slower.
1663 // generally speaking, the cost of bigger code size is usually not
1664 // worth the trade-off since larger code-size will hinder inlinability
1665 // but depending on your needs, you may want to uncomment the pragma
1666 // below to unroll the loop.
1667 //#pragma GCC unroll 2
1668 for (; l >= 32; l -= 32) {
1669 for (int i = 0; i < 4; ++i, p += 8) {
1670 uint64_t stripe = bserial__chibihash64__load64le(p);
1671 h[i] = (stripe + h[i]) * K;
1672 h[(i+1)&3] += bserial__chibihash64__rotl(stripe, 27);
1673 }
1674 }
1675
1676 for (; l >= 8; l -= 8, p += 8) {
1677 h[0] ^= bserial__chibihash64__load32le(p+0); h[0] *= K;
1678 h[1] ^= bserial__chibihash64__load32le(p+4); h[1] *= K;
1679 }
1680
1681 if (l >= 4) {
1682 h[2] ^= bserial__chibihash64__load32le(p);
1683 h[3] ^= bserial__chibihash64__load32le(p + l - 4);
1684 } else if (l > 0) {
1685 h[2] ^= p[0];
1686 h[3] ^= p[l/2] | ((uint64_t)p[l-1] << 8);
1687 }
1688
1689 h[0] += bserial__chibihash64__rotl(h[2] * K, 31) ^ (h[2] >> 31);
1690 h[1] += bserial__chibihash64__rotl(h[3] * K, 31) ^ (h[3] >> 31);
1691 h[0] *= K; h[0] ^= h[0] >> 31;
1692 h[1] += h[0];
1693
1694 uint64_t x = (uint64_t)len * K;
1695 x ^= bserial__chibihash64__rotl(x, 29);
1696 x += seed;
1697 x ^= h[1];
1698
1699 x ^= bserial__chibihash64__rotl(x, 15) ^ bserial__chibihash64__rotl(x, 42);
1700 x *= K;
1701 x ^= bserial__chibihash64__rotl(x, 13) ^ bserial__chibihash64__rotl(x, 31);
1702
1703 return x;
1704}
1705
1706// }}}

1708static inline uint64_t
1709bserial_hash(const void* data, size_t len) {
1710 return bserial__chibihash64(data, (ptrdiff_t)len, 0);
1711}

1714static inline int32_t
1715bserial_lookup_index(uint64_t hash, int32_t exp, int32_t idx) {
1716 uint32_t mask = ((uint32_t)1 << exp) - 1;
1717 uint32_t step = (uint32_t)((hash >> (64 - exp)) | 1);
1718 return (idx + step) & mask;
1719}

1721static inline bserial_status_t
1722bserial_skip_next(bserial_ctx_t* ctx, uint32_t depth) {
1723 uint8_t marker;
1724 BSERIAL_CHECK_STATUS(bserial_peek_marker(ctx, &marker));
1725
1726 switch ((bserial_marker_t)marker) {
1727 case BSERIAL_UINT:
1728 {
1729 uint64_t u64;
1730 BSERIAL_CHECK_STATUS(bserial_uint(ctx, &u64));
1731 }
1732 break;
1733 case BSERIAL_SINT:
1734 {
1735 int64_t s64;
1736 BSERIAL_CHECK_STATUS(bserial_sint(ctx, &s64));
1737 }
1738 break;
1739 case BSERIAL_F32:
1740 {
1741 bserial_discard_marker(ctx);
1742 BSERIAL_CHECK_STATUS(ctx->status = bserial_skip(ctx->in, sizeof(float)));
1743 }
1744 break;
1745 case BSERIAL_F64:
1746 {
1747 bserial_discard_marker(ctx);
1748 BSERIAL_CHECK_STATUS(ctx->status = bserial_skip(ctx->in, sizeof(double)));
1749 }
1750 break;
1751 case BSERIAL_BLOB:
1752 {
1753 bserial_discard_marker(ctx);
1754 uint64_t len;
1755 BSERIAL_CHECK_STATUS(bserial_read_uint(&len, ctx->in));
1756 BSERIAL_CHECK_STATUS(ctx->status = bserial_skip(ctx->in, len));
1757 }
1758 break;
1759 case BSERIAL_SYM_DEF:
1760 case BSERIAL_SYM_REF:
1761 {
1762 const char* sym;
1763 uint64_t sym_len;
1764 BSERIAL_CHECK_STATUS(bserial_symbol(ctx, &sym, &sym_len));
1765 }
1766 break;
1767 case BSERIAL_ARRAY:
1768 {
1769 bserial_discard_marker(ctx);
1770
1771 uint64_t len;
1772 BSERIAL_CHECK_STATUS(bserial_read_uint(&len, ctx->in));
1773 if (len > 0 && depth == 0) { return bserial_malformed(ctx); }
1774
1775 for (uint64_t i = 0; i < len; ++i) {
1776 BSERIAL_CHECK_STATUS(bserial_skip_next(ctx, depth - 1));
1777 }
1778 }
1779 break;
1780 case BSERIAL_TABLE:
1781 {
1782 bserial_discard_marker(ctx);
1783
1784 uint64_t num_rows;
1785 BSERIAL_CHECK_STATUS(bserial_read_uint(&num_rows, ctx->in));
1786 if (num_rows > 0 && depth == 0) { return bserial_malformed(ctx); }
1787
1788 if (num_rows > 0) {
1789 uint64_t num_cols;
1790 BSERIAL_CHECK_STATUS(bserial_read_uint(&num_cols, ctx->in));
1791
1792 for (uint64_t i = 0; i < num_cols; ++i) {
1793 const char* sym;
1794 uint64_t sym_len;
1795 BSERIAL_CHECK_STATUS(bserial_symbol(ctx, &sym, &sym_len));
1796 }
1797
1798 for (uint64_t i = 0; i < num_rows; ++i) {
1799 for (uint64_t j = 0; j < num_cols; ++j) {
1800 BSERIAL_CHECK_STATUS(bserial_skip_next(ctx, depth - 1));
1801 }
1802 }
1803 }
1804 }
1805 break;
1806 case BSERIAL_RECORD:
1807 {
1808 bserial_discard_marker(ctx);
1809
1810 uint64_t num_cols;
1811 BSERIAL_CHECK_STATUS(bserial_read_uint(&num_cols, ctx->in));
1812
1813 for (uint64_t i = 0; i < num_cols; ++i) {
1814 const char* sym;
1815 uint64_t sym_len;
1816 BSERIAL_CHECK_STATUS(bserial_symbol(ctx, &sym, &sym_len));
1817 }
1818
1819 for (uint64_t i = 0; i < num_cols; ++i) {
1820 BSERIAL_CHECK_STATUS(bserial_skip_next(ctx, depth - 1));
1821 }
1822 }
1823 break;
1824 default:
1825 return bserial_malformed(ctx);
1826 }
1827
1828 return BSERIAL_OK;
1829}

1831static inline bool
1832bserial_probe_next_record_field(bserial_ctx_t* ctx) {
1833 bserial_scope_t* scope = ctx->scope;
1834
1835 while (scope->iterator < scope->len) {
1836 if (scope->record_schema[scope->iterator].field_name != NULL) {
1837 return true;
1838 } else {
1839 if (
1840 bserial_skip_next(
1841 ctx,
1842 ctx->config.max_depth - (uint32_t)(ctx->scope - ctx->scope_first)
1843 ) != BSERIAL_OK
1844 ) {
1845 return false;
1846 }
1847 ++scope->iterator;
1848 }
1849 }
1850
1851 bserial_end_op(ctx, BSERIAL_OP_RECORD);
1852 return false;

1855bserial_status_t
1856bserial_symbol(bserial_ctx_t* ctx, const char** buf, uint64_t* len) {
1857 BSERIAL_CHECK_STATUS(bserial_begin_op(ctx, BSERIAL_OP_SYMBOL));
1858
1859 if (bserial_mode(ctx) == BSERIAL_MODE_READ) {
1860 uint8_t marker;
1861 BSERIAL_CHECK_STATUS(bserial_read_marker(ctx, &marker));
1862
1863 if (marker == BSERIAL_SYM_DEF) {
1864 if (ctx->num_symbols >= ctx->config.max_num_symbols) { return bserial_malformed(ctx); }
1865
1866 uint64_t symbol_len;
1867 BSERIAL_CHECK_STATUS(ctx->status = bserial_read_uint(&symbol_len, ctx->in));
1868 if (symbol_len > ctx->config.max_symbol_len) { return bserial_malformed(ctx); }
1869
1870 BSERIAL_CHECK_STATUS(ctx->status = bserial_read(ctx->in, ctx->strpool, symbol_len));
1871 ctx->strpool[symbol_len] = '\0';
1872
1873 ctx->symtab[ctx->num_symbols] = (bserial_symbol_t){
1874 .buf = ctx->strpool,
1875 .len = symbol_len,
1876 };
1877
1878 *buf = ctx->strpool;
1879 *len = symbol_len;
1880
1881 ctx->num_symbols += 1;
1882 ctx->strpool += (symbol_len + 1);
1883 } else if (marker == BSERIAL_SYM_REF) {
1884 uint64_t symbol_id;
1885 BSERIAL_CHECK_STATUS(ctx->status = bserial_read_uint(&symbol_id, ctx->in));
1886 if (symbol_id >= ctx->num_symbols) { return bserial_malformed(ctx); }
1887
1888 *buf = ctx->symtab[symbol_id].buf;
1889 *len = ctx->symtab[symbol_id].len;
1890 } else {
1891 return bserial_malformed(ctx);
1892 }
1893 } else {
1894 uint64_t symbol_len = *len;
1895 if (symbol_len > ctx->config.max_symbol_len) { return bserial_malformed(ctx); }
1896
1897 uint64_t symbol_hash = bserial_hash(*buf, symbol_len);
1898 for (int32_t i = (int32_t)symbol_hash;;) {
1899 i = bserial_lookup_index(symbol_hash, ctx->symtab_exp, i);
1900 int32_t index = ctx->symtab_index[i];
1901 if (index == 0) {
1902 if (ctx->num_symbols >= ctx->config.max_num_symbols) {
1903 return bserial_malformed(ctx);
1904 }
1905 memcpy(ctx->strpool, *buf, symbol_len);
1906 ctx->strpool[symbol_len] = '\0';
1907
1908 ctx->symtab[ctx->num_symbols] = (bserial_symbol_t){
1909 .buf = ctx->strpool,
1910 .len = symbol_len,
1911 };
1912 ctx->symtab_index[i] = ctx->num_symbols + 1;
1913 ctx->num_symbols += 1;
1914 ctx->strpool += (symbol_len + 1);
1915
1916 uint8_t marker = BSERIAL_SYM_DEF;
1917 BSERIAL_CHECK_STATUS(ctx->status = bserial_write(ctx->out, &marker, sizeof(marker)));
1918 BSERIAL_CHECK_STATUS(ctx->status = bserial_write_uint(symbol_len, ctx->out));
1919 BSERIAL_CHECK_STATUS(ctx->status = bserial_write(ctx->out, *buf, symbol_len));
1920
1921 break;
1922 } else if (
1923 ctx->symtab[index - 1].len == symbol_len
1924 && memcmp(ctx->symtab[index - 1].buf, *buf, symbol_len) == 0
1925 ) {
1926 uint8_t marker = BSERIAL_SYM_REF;
1927 BSERIAL_CHECK_STATUS(ctx->status = bserial_write(ctx->out, &marker, sizeof(marker)));
1928 BSERIAL_CHECK_STATUS(ctx->status = bserial_write_uint((uint64_t)index - 1, ctx->out));

1967
1968 bserial_scope_t* scope = ctx->scope;
1969
1970 if (bserial_mode(ctx) == BSERIAL_MODE_READ) {
1971 if (scope->type == BSERIAL_SCOPE_RECORD && record == scope->record_addr) {
1972 switch (scope->record_mode) {
1973 case BSERIAL_RECORD_KEY_IO:
1974 scope->record_mode = BSERIAL_RECORD_VALUE_IO;
1975 scope->iterator = 0;
1976 return bserial_probe_next_record_field(ctx);
1977 case BSERIAL_RECORD_VALUE_IO:
1978 if (scope->iterator == scope->len) {
1979 bserial_end_op(ctx, BSERIAL_OP_RECORD);
1980 return false;
1981 } else {
1982 return bserial_probe_next_record_field(ctx);
1983 }
1984 default:
1985 bserial_malformed(ctx);
1986 return false;
1987 }
1988 } else {
1989 bserial_scope_t* parent_scope = scope;
1990 if (bserial_begin_op(ctx, BSERIAL_OP_RECORD) != BSERIAL_OK) {
1991 return false;
1992 }
1993 scope = ctx->scope;
1994 scope->record_addr = record;
1995
1996 if (parent_scope->type != BSERIAL_SCOPE_TABLE) {
1997 uint8_t marker;
1998 if ((bserial_read_marker(ctx, &marker)) != BSERIAL_OK) {
1999 return false;
2000 }
2001 if (marker != BSERIAL_RECORD) {
2002 bserial_malformed(ctx);
2003 return false;
2004 }
2005 }
2006
2007 // In case of a table, allocate schema at the table's scope.
2008 bserial_scope_t* schema_scope =
2009 parent_scope->type == BSERIAL_SCOPE_TABLE
2010 ? parent_scope
2011 : scope;
2012
2013 // Schema discovery only happens for the first row of a table
2014 if (parent_scope->type != BSERIAL_SCOPE_TABLE || parent_scope->iterator == 1) {
2015 scope->record_mode = BSERIAL_RECORD_KEY_IO;
2016
2017 uint64_t num_fields;
2018 if ((ctx->status = bserial_read_uint(&num_fields, ctx->in)) != BSERIAL_OK) {
2019 return false;
2020 }
2021 if (num_fields > ctx->config.max_record_fields) {
2022 bserial_malformed(ctx);
2023 return false;
2024 }
2025
2026 for (uint64_t i = 0; i < num_fields; ++i) {
2027 schema_scope->record_schema[i].field_name = NULL;
2028 }
2029
2030 schema_scope->record_width = num_fields;
2031 for (uint64_t i = 0; i < num_fields; ++i) {
2032 const char* symbol;
2033 uint64_t symbol_len;
2034 scope->iterator = i;
2035 if (bserial_symbol(ctx, &symbol, &symbol_len) != BSERIAL_OK) {
2036 return false;
2037 }
2038
2039 schema_scope->record_schema[i].symbol = symbol;
2040 schema_scope->record_schema[i].symbol_len = symbol_len;
2041 }
2042 scope->iterator = 0;
2043 } else {
2044 scope->record_mode = BSERIAL_RECORD_VALUE_IO;
2045 }
2046
2047 scope->record_schema = schema_scope->record_schema;
2048 scope->len = schema_scope->record_width;
2049
2050 if (scope->record_mode == BSERIAL_RECORD_VALUE_IO) {
2051 return bserial_probe_next_record_field(ctx);
2052 } else {
2053 return true;
2054 }
2055 }
2056 } else {
2057 if (scope->type == BSERIAL_SCOPE_RECORD && scope->record_addr == record) {
2058 switch (scope->record_mode) {
2059 case BSERIAL_RECORD_MEASURE_WIDTH:
2060 scope->record_mode = BSERIAL_RECORD_KEY_IO;
2061 ctx->status = bserial_write_uint(scope->len, ctx->out);
2062 return ctx->status == BSERIAL_OK;
2063 case BSERIAL_RECORD_KEY_IO:
2064 scope->record_mode = BSERIAL_RECORD_VALUE_IO;
2065 scope->iterator = 0;
2066 return true;
2067 case BSERIAL_RECORD_VALUE_IO:
2068 bserial_end_op(ctx, BSERIAL_OP_RECORD);
2069 return false;
2070 default:
2071 bserial_malformed(ctx);
2072 return false;
2073 }
2074 } else {
2075 bserial_scope_t* parent_scope = scope;
2076 if (bserial_begin_op(ctx, BSERIAL_OP_RECORD) != BSERIAL_OK) {
2077 return false;
2078 }
2079 scope = ctx->scope;
2080 scope->record_addr = record;
2081
2082 if (parent_scope->type != BSERIAL_SCOPE_TABLE || parent_scope->iterator == 1) {
2083 scope->record_mode = BSERIAL_RECORD_MEASURE_WIDTH;
2084
2085 if (parent_scope->type != BSERIAL_SCOPE_TABLE) {
2086 uint8_t marker = BSERIAL_RECORD;
2087 if ((ctx->status = bserial_write(ctx->out, &marker, sizeof(marker))) != BSERIAL_OK) {
2088 return false;
2089 }
2090 }
2091 } else {
2092 scope->record_mode = BSERIAL_RECORD_VALUE_IO;
2093 }
2094
2095 return true;
2096 }
2097 }
