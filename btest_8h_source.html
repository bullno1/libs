341bool
342btest_debugger_attached(void) {
343 // TracerPid is non-zero while a process is being traced.
344 // It sits near the top of the file so a single read always covers it.
345 int fd = open("/proc/self/status", O_RDONLY);
346 if (fd < 0) { return false; }
347
348 char status[1024];
349 ssize_t num_bytes_read = read(fd, status, sizeof(status) - 1);
350 close(fd);
351 if (num_bytes_read <= 0) { return false; }
352 status[num_bytes_read] = '\0';
353
354 const char* tracer_pid = strstr(status, "TracerPid:");
355 if (tracer_pid == NULL) { return false; }
356
357 tracer_pid += sizeof("TracerPid:") - 1;
358 while (*tracer_pid == ' ' || *tracer_pid == '\t') { ++tracer_pid; }
359
360 return *tracer_pid != '0';
361}
