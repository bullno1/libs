366 btest__ctx.current_suite->cleanup_per_suite();

371 if (btest__ctx.current_suite->init_per_suite) {

377 test->suite->init_per_test();

380 btest__ctx.success = true;

384 test->suite->cleanup_per_test();

416bool
417btest_debugger_attached(void) {
418 // TracerPid is non-zero while a process is being traced.
419 // It sits near the top of the file so a single read always covers it.
420 int fd = open("/proc/self/status", O_RDONLY);
421 if (fd < 0) { return false; }
422
423 char status[1024];
424 ssize_t num_bytes_read = read(fd, status, sizeof(status) - 1);
425 close(fd);
426 if (num_bytes_read <= 0) { return false; }
427 status[num_bytes_read] = '\0';
428
429 const char* tracer_pid = strstr(status, "TracerPid:");
430 if (tracer_pid == NULL) { return false; }
431
432 tracer_pid += sizeof("TracerPid:") - 1;
433 while (*tracer_pid == ' ' || *tracer_pid == '\t') { ++tracer_pid; }
434
435 return *tracer_pid != '0';
436}
