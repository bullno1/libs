299static barg_result_t
300barg_try_parse(
301 const char* arg,
302 barg_opt_t* opt,
303 int argc, const char* argv[],
304 int arg_index, bool is_long
305) {
306 if (!opt->repeatable && opt->barg__count >= 1) {
307 return BARG_MAKE_ERROR("Option can only be specified once");
308 }
309
310 int opt_name_len = is_long ? (int)opt->barg__name_len : 1;
311 char separator = arg[opt_name_len];
312 if (opt->boolean) {
313 if (separator == '\0') {
314 if (opt->parser.userdata == &BARG_HELP_MAGIC) {
315 return (barg_result_t) {
316 .status = BARG_SHOW_HELP,
317 .arg_index = arg_index,
318 };
319 }
320
321 const char* error = opt->parser.parse(opt->parser.userdata, NULL);
322
323 if (error == NULL) {
324 ++opt->barg__count;
325 return (barg_result_t){
326 .status = BARG_OK,
327 .arg_index = arg_index,
328 };
329 } else {
330 return BARG_MAKE_ERROR(error);
331 }
332 } else {
333 return BARG_MAKE_ERROR("Invalid usage of a boolean flag");
334 }
335 } else {
336 const char* value;
337 if (separator == '=') { // --opt=value
338 value = arg + opt_name_len + 1;
339 } else if (separator == '\0') { // --opt value
340 if ((arg_index + 1) < argc) { // Grab the next arg
341 value = argv[++arg_index];
342 } else {
343 return BARG_MAKE_ERROR("Option must be followed by a value");
344 }
345 } else { //--optsomethingelse
346 if (is_long) {
347 return BARG_MAKE_ERROR("Unknown option");
348 } else { // -ovalue
349 value = &arg[1];
350 }
351 }
352
353 const char* error = opt->parser.parse(opt->parser.userdata, value);
354 if (error == NULL) {
355 ++opt->barg__count;
356 return (barg_result_t){
357 .status = BARG_OK,
358 .arg_index = arg_index,
359 };
360 } else {
361 return (barg_result_t){
362 .status = BARG_PARSE_ERROR,
363 .arg_index = arg_index,
364 .value = value,
365 .message = error,
366 };
367 }
368 }

383barg_result_t
384barg_parse(barg_t* barg, int argc, const char* argv[]) {
385 int num_opts = barg->num_opts;
386 barg_opt_t* opts = barg->opts;
387 for (int i = 0; i < num_opts; ++i) {
388 opts[i].barg__count = 0;
389 if (opts[i].name != NULL) {
390 opts[i].barg__name_len = strlen(opts[i].name);
391 } else {
392 opts[i].barg__name_len = 0;
393 }
394 }
395
396 int arg_index = 1;
397 for (; arg_index < argc; ++arg_index) {
398 const char* arg = argv[arg_index];
399 if (arg[0] == '-') {
400 if (arg[1] == '-') {
401 if (arg[2] == '\0') { // End of options
402 return barg_handle_positional(barg, ++arg_index, argc, argv);
403 } else { // Long name
404 arg += 2; // Skip '--'
405 barg_opt_t* opt = NULL;
406 // Find a matching opt
407 for (int opt_index = 0; opt_index < num_opts; ++opt_index) {
408 const char* name = opts[opt_index].name;
409 size_t name_len = opts[opt_index].barg__name_len;
410 size_t arg_len = strlen(arg);
411
412 if (arg_len >= name_len && memcmp(name, arg, name_len) == 0) {
413 opt = &opts[opt_index];
414 break;
415 }
416 }
417
418 if (opt == NULL) {
419 return BARG_MAKE_ERROR("Unknown option");
420 }
421
422 barg_result_t result = barg_try_parse(
423 arg, opt,
424 argc, argv,
425 arg_index, true
426 );
427
428 if (result.status == BARG_OK) {
429 arg_index = result.arg_index;
430 } else {
431 return result;
432 }
433 }
434 } else { // Short name
435 arg += 1; // Skip '-'
436 barg_opt_t* opt = NULL;
437 // Find a matching opt
438 for (int opt_index = 0; opt_index < num_opts; ++opt_index) {
439 char short_name = opts[opt_index].short_name;
440 if (short_name != '\0' && short_name == arg[0]) {
441 opt = &opts[opt_index];
442 break;
443 }
444 }
445
446 if (opt == NULL) {
447 return BARG_MAKE_ERROR("Unknown option");
448 }
449
450 barg_result_t result = barg_try_parse(
451 arg, opt,
452 argc, argv,
453 arg_index, false
454 );
455
456 if (result.status == BARG_OK) {
457 arg_index = result.arg_index;
458 } else {
459 return result;
460 }
461 }
462 } else { // Positional
463 return barg_handle_positional(barg, arg_index, argc, argv);
464 }
465 }
466
467 return barg_handle_positional(barg, arg_index, argc, argv);
468}

566static void
567barg_print_help(barg_t* barg, FILE* file) {
568 if (barg->usage != NULL) {
569 fprintf(file, "Usage: %s\n", barg->usage);
570 }
571
572 if (barg->summary != NULL) {
573 if (barg->usage != NULL) {
574 fprintf(file, "\n");
575 }
576 fprintf(file, "%s\n", barg->summary);
577 }
578
579 bool printed_header = false;
580 for (int i = 0; i < barg->num_opts; ++i) {
581 const barg_opt_t* opt = &barg->opts[i];
582 if (opt->hidden) { continue; }
583
584 if (!printed_header) {
585 if (barg->summary != NULL) {
586 fprintf(file, "\n");
587 }
588 fprintf(file, "Options:\n");
589 printed_header = true;
590 }
591
592 fprintf(file, "\n");
593
594 if (opt->short_name != 0) {
595 fprintf(file, "-%c", opt->short_name);
596 }
597
598 if (opt->name != NULL) {
599 if (opt->short_name != 0) {
600 fprintf(file, ", ");
601 }
602 fprintf(file, "--%s", opt->name);
603 }
604
605 if (!opt->boolean) {
606 fprintf(
607 file, "=<%s>",
608 opt->value_name != NULL ? opt->value_name : "value"
609 );
610 }
611
612 if (opt->summary != NULL) {
613 fprintf(file, ": %s", opt->summary);
614 }
615
616 fprintf(file, "\n");
617
618 if (opt->description != NULL) {
619 fprintf(file, "\n");
620 const char* line_start = opt->description;
621 while (*line_start != '\0') {
622 const char* line_end = line_start;
623 while (*line_end != '\n' && *line_end != '\0') {
624 ++line_end;
625 }
626
627 fprintf(file, " %.*s\n", (int)(line_end - line_start), line_start);
628 line_start = *line_end == '\0' ? line_end : line_end + 1;
629 }
630 }
631 }
