866// State that only the root of a bco_call chain needs.
867// It sits at the start of the root's stack instead of in every header.
868//
869// Only the innermost coroutine of a chain can be waiting for a value, the rest
870// are parked in their bco_call, so one wait per root is enough.
871typedef struct {

893bco_status_t
894bco_resume(bco_t* coro) {
895 if (coro->status != BCO_SUSPENDED) { return coro->status; }
896 BCO_ASSERT(!coro->relocating, "Coroutine was resumed between bco_reload_begin and bco_reload_end");
897 // A chain waiting for a value has nothing to do until it arrives.
898 // Termination still has to get through to run the cleanup section.
899 if (coro->resume_point != -1 && bco__is_receiving(coro)) { return BCO_SUSPENDED; }
900
901 coro->status = BCO_RUNNING;
902 coro->sp = coro->bp;
903 coro->fn(coro, coro->args);
904 BCO_ASSERT(coro->resume_point != 0, "Bare return was used to return from coroutine");
905 return coro->status = coro->resume_point == -1 ? BCO_TERMINATED : BCO_SUSPENDED;
906}

997bco__spawn(bco_t* coro, bco_t* parent, bco_fn_t fn, size_t args_size, size_t args_alignment, void* args) {
998 coro->resume_point = 0;
999 coro->relocating = false;
1000 coro->status = BCO_SUSPENDED;
1001 coro->subcoro = NULL;
1002 coro->userdata = NULL;
1003 coro->sp = coro->stack;
1004 if (parent == NULL) {
1005 coro->root = coro;
1006 bco__root_t* root = bco__alloc(coro, sizeof(bco__root_t), _Alignof(bco__root_t));
1007 memset(root, 0, sizeof(*root));
1008 } else {
1009 coro->root = parent->root;
1010 }
1011 coro->fn = fn;
1012 coro->args = bco__alloc(coro, args_size, args_alignment);
1013 coro->bp = coro->sp;
1014 memcpy(coro->args, args, args_size);
1015}

1048void
1049bco__on_terminate(bco_t* coro) {
1050 coro->resume_point = -1;
1051
1052 // Every exit path ends here, so this is the one place to cancel a pending
1053 // receive. Without it a parent carrying on after its subcoroutine was
1054 // terminated by bco_reload_end would leave a stale destination pointer behind.
1055 bco__root_t* root = bco__root(coro);
1056 if (root->receiver == coro) {
1057 root->receiver = NULL;
1058 root->recv_ready = false;
1059 }
1060}

1094bool
1095bco__send(bco_t* coro, const char* type_name, size_t size, const void* value) {
1096 BCO_ASSERT(coro->root == coro, "bco_send must be given the handle passed to bco_spawn");
1097 if (coro->status != BCO_SUSPENDED) { return false; }
1098
1099 bco__root_t* root = bco__root(coro);
1100 if (root->receiver == NULL || root->recv_ready) { return false; }
1101
1102 if (root->recv_type != NULL) {
1103 // Both literals usually come from the same build but merging them is
1104 // not guaranteed across translation units
1105 if (root->recv_type != type_name && strcmp(root->recv_type, type_name) != 0) { return false; }
1106 BCO_ASSERT(root->recv_size == size, "Sent type is defined differently from the received one");
1107 } else {
1108 // The receiver's literal was unloaded by a reload.

1172 coro->relocate_found = true;
