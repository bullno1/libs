553bco_status_t
554bco_resume(bco_t* coro) {
555 if (coro->status != BCO_SUSPENDED) { return coro->status; }
556
557 coro->status = BCO_RUNNING;
558 coro->sp = coro->bp;
559 coro->fn(coro, coro->args);
560 BCO_ASSERT(coro->resume_point != 0, "Bare return was used to return from coroutine");
561 return coro->status = coro->resume_point >= 0 ? BCO_SUSPENDED : BCO_TERMINATED;
562}

574void
575bco_terminate(bco_t* coro) {
576 if (coro->status != BCO_SUSPENDED) { return; }
577 if (coro->resume_point == 0) {
578 coro->status = BCO_TERMINATED;
579 return;

587 coro->resume_point = -1;
588 bco_resume(coro);
