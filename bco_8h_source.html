749bco_resume(bco_t* coro) {
750 if (coro->status != BCO_SUSPENDED) { return coro->status; }
751 BCO_ASSERT(!coro->relocating, "Coroutine was resumed between bco_reload_begin and bco_reload_end");
752

754 coro->sp = coro->bp;
755 coro->fn(coro, coro->args);
756 BCO_ASSERT(coro->resume_point != 0, "Bare return was used to return from coroutine");

895bco__hash_next_name(const char** cursor) {
896 const char* s = *cursor;
897 while (*s != '\0' && !bco__is_ident(*s)) { ++s; }

913void
914bco__relocate(bco_t* coro, const char* names, int count) {
915 if (!coro->relocating) {
916 // Begin: the old code is still loaded, hash only the name we are parked at
917 int index = -(int)coro->named_point - BCO__NAMED_FIRST;
918 unsigned int h = 0;
919 for (int i = 0; i <= index; ++i) { h = bco__hash_next_name(&names); }
920 coro->named_point = h;
921 coro->relocate_found = true;
922 return;
923 }
924
925 // End: find the same name in the new code
926 for (int i = 0; i < count; ++i) {
927 if (bco__hash_next_name(&names) == coro->named_point) {
928 coro->resume_point = -(BCO__NAMED_FIRST + i);
929 coro->relocate_found = true;
930 return;
931 }
932 }
933 coro->relocate_found = false;
934}

954bool
955bco_reload_begin(bco_t* coro) {
956 if (!bco_reloadable(coro)) { return false; }
957 if (coro->status != BCO_SUSPENDED || coro->resume_point == 0) { return true; }
958 BCO_ASSERT(!coro->relocating, "bco_reload_begin was called twice");
959
960 int resume_point = coro->resume_point;
961 coro->named_point = (unsigned int)resume_point; // Tell bco__relocate which name to hash
962 bco__relocate_call(coro);
963 coro->resume_point = resume_point;
964 coro->relocating = true;
965
966 return coro->subcoro == NULL || bco_reload_begin(coro->subcoro);
967}

969bool
970bco_reload_end(bco_t* coro) {
971 if (coro->status != BCO_SUSPENDED || coro->resume_point == 0) { return true; }
972 BCO_ASSERT(coro->relocating, "bco_reload_end was called without bco_reload_begin");
973
974 bco__relocate_call(coro);
975 coro->relocating = false;
976 if (!coro->relocate_found) {
977 coro->resume_point = BCO__RELOCATE; // Anything but 0 so bco_terminate runs the cleanup
978 bco_terminate(coro);
979 return false;
980 }
981
982 return coro->subcoro == NULL || bco_reload_end(coro->subcoro);
983}
