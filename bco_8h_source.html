872// Hash the next identifier in a stringified list of names, advancing past it
873static unsigned int
874bco__hash_next_name(const char** cursor) {
875 const char* s = *cursor;
876 while (*s != '\0' && !bco__is_ident(*s)) { ++s; }
877
878 unsigned int h = 2166136261u; // FNV-1a
879 for (; bco__is_ident(*s); ++s) {
880 h = (h ^ (unsigned char)*s) * 16777619u;

892void
893bco__relocate(bco_t* coro, const char* names, int count) {
894 if (!coro->relocating) {
895 // Begin: the old code is still loaded, hash only the name we are parked at
896 int index = -(int)coro->named_point - BCO__NAMED_FIRST;
897 unsigned int h = 0;
898 for (int i = 0; i <= index; ++i) { h = bco__hash_next_name(&names); }
899 coro->named_point = h;
900 coro->relocate_found = true;
901 return;
902 }
903
904 // End: find the same name in the new code
905 for (int i = 0; i < count; ++i) {
906 if (bco__hash_next_name(&names) == coro->named_point) {
907 coro->resume_point = -(BCO__NAMED_FIRST + i);
908 coro->relocate_found = true;
909 return;
910 }
911 }
912 coro->relocate_found = false;
913}

933bool
934bco_reload_begin(bco_t* coro) {
935 if (!bco_reloadable(coro)) { return false; }
936 if (coro->status != BCO_SUSPENDED || coro->resume_point == 0) { return true; }
937 BCO_ASSERT(!coro->relocating, "bco_reload_begin was called twice");
938

940 coro->named_point = (unsigned int)resume_point; // Tell bco__relocate which name to hash
941 bco__relocate_call(coro);
942 coro->resume_point = resume_point;

948bool
949bco_reload_end(bco_t* coro) {
950 if (coro->status != BCO_SUSPENDED || coro->resume_point == 0) { return true; }
951 BCO_ASSERT(coro->relocating, "bco_reload_end was called without bco_reload_begin");
952
953 bco__relocate_call(coro);

957 bco_terminate(coro);
958 return false;
959 }
