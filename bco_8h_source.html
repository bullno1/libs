1087bco_status_t
1088bco_resume(bco_t* coro) {
1089 if (coro->status != BCO_SUSPENDED) { return coro->status; }
1090 BCO_ASSERT(!coro->relocating, "Coroutine was resumed between bco_reload_begin and bco_reload_end");
1091 // A chain waiting for a value has nothing to do until it arrives.
1092 // Termination still has to get through to run the cleanup section.
1093 if (coro->resume_point != -1 && bco__is_receiving(coro)) { return BCO_SUSPENDED; }
1094
1095 coro->status = BCO_RUNNING;
1096 coro->sp = coro->bp;
1097 coro->fn(coro, coro->args);
1098 BCO_ASSERT(coro->resume_point != 0, "Bare return was used to return from coroutine");
1099 return coro->status = coro->resume_point == -1 ? BCO_TERMINATED : BCO_SUSPENDED;
1100}

1307bool
1308bco__send(bco_t* coro, const char* type_name, size_t size, const void* value) {
1309 BCO_ASSERT(coro->root == coro, "bco_send must be given the handle passed to bco_spawn");
1310 if (coro->status != BCO_SUSPENDED) { return false; }
1311
1312 bco__root_t* root = bco__root(coro);
1313 if (root->receiver == NULL || root->recv_ready) { return false; }
1314
1315 if (root->recv_type != NULL) {
1316 // Both literals usually come from the same build but merging them is
1317 // not guaranteed across translation units
1318 if (root->recv_type != type_name && strcmp(root->recv_type, type_name) != 0) { return false; }
1319 BCO_ASSERT(root->recv_size == size, "Sent type is defined differently from the received one");
1320 } else {
1321 // The receiver's literal was unloaded by a reload.
1322 // Its new build reads whatever the new build of the sender writes, so
1323 // the size recorded by the old build no longer applies.
1324 if (bco__hash_str(type_name) != root->recv_type_hash) { return false; }
1325 root->recv_size = size;
1326 root->recv_type = type_name; // Back on the fast path
1327 // The variable may have grown past the frame the old build allocated.
1328 // The new build reallocates the frame on the next resume, but a copy
1329 // taken before that must already cover the whole value.
1330 char* end = (char*)root->recv_at + size;
1331 if (end > root->receiver->sp) { root->receiver->sp = end; }
1332 }
1333
1334 memcpy(root->recv_at, value, size);
1335 root->recv_ready = true;
1336 return true;
1337}

1374void
1375bco__relocate(bco_t* coro, const char* names, int count) {
1376 if (!coro->relocating) {
1377 // Begin: the old code is still loaded, hash only the name we are parked at
1378 int index = -(int)coro->named_point - BCO__NAMED_FIRST;
1379 unsigned int h = 0;
1380 for (int i = 0; i <= index; ++i) { h = bco__hash_next_name(&names); }
1381 coro->named_point = h;
1382 coro->relocate_found = true;
1383 return;
1384 }
1385
1386 // End: find the same name in the new code
1387 for (int i = 0; i < count; ++i) {
1388 if (bco__hash_next_name(&names) == coro->named_point) {
1389 coro->resume_point = -(BCO__NAMED_FIRST + i);
1390 coro->relocate_found = true;
1391 return;
1392 }
1393 }
1394 coro->relocate_found = false;
1395}

1421bool
1422bco_reload_begin(bco_t* coro) {
1423 if (!bco_reloadable(coro, NULL)) { return false; }
1424 if (coro->status != BCO_SUSPENDED || coro->resume_point == 0) { return true; }
1425 BCO_ASSERT(!coro->relocating, "bco_reload_begin was called twice");
1426
1427 // The type name literal of a pending receive is about to be unloaded with
1428 // the old code, keep its hash instead. Only the root holds it.
1429 if (coro->root == coro) {
1430 bco__root_t* root = bco__root(coro);
1431 if (root->receiver != NULL && root->recv_type != NULL) {
1432 root->recv_type_hash = bco__hash_str(root->recv_type);
1433 root->recv_type = NULL;
1434 }
1435 }
1436
1437 int resume_point = coro->resume_point;
1438 coro->named_point = (unsigned int)resume_point; // Tell bco__relocate which name to hash
1439 bco__relocate_call(coro);
1440 coro->resume_point = resume_point;
1441 coro->relocating = true;
1442
1443 return coro->subcoro == NULL || bco_reload_begin(coro->subcoro);
1444}

1448 if (coro->status != BCO_SUSPENDED || coro->resume_point == 0) { return true; }
1449 BCO_ASSERT(coro->relocating, "bco_reload_end was called without bco_reload_begin");

1452 coro->relocating = false;

1454 coro->resume_point = BCO__RELOCATE; // Anything but 0 so bco_terminate runs the cleanup
1455 bco_terminate(coro);
1456 return false;
1457 }
