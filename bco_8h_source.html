926void
927bco__relocate(bco_t* coro, const char* names, int count) {
928 if (!coro->relocating) {
929 // Begin: the old code is still loaded, hash only the name we are parked at
930 int index = -(int)coro->named_point - BCO__NAMED_FIRST;
931 unsigned int h = 0;
932 for (int i = 0; i <= index; ++i) { h = bco__hash_next_name(&names); }
933 coro->named_point = h;
934 coro->relocate_found = true;
935 return;
936 }
937
938 // End: find the same name in the new code
939 for (int i = 0; i < count; ++i) {
940 if (bco__hash_next_name(&names) == coro->named_point) {
941 coro->resume_point = -(BCO__NAMED_FIRST + i);
942 coro->relocate_found = true;
943 return;
944 }
945 }
946 coro->relocate_found = false;
947}

967bool
968bco_reload_begin(bco_t* coro) {
969 if (!bco_reloadable(coro)) { return false; }
970 if (coro->status != BCO_SUSPENDED || coro->resume_point == 0) { return true; }
971 BCO_ASSERT(!coro->relocating, "bco_reload_begin was called twice");
972
973 int resume_point = coro->resume_point;
974 coro->named_point = (unsigned int)resume_point; // Tell bco__relocate which name to hash
975 bco__relocate_call(coro);
976 coro->resume_point = resume_point;
977 coro->relocating = true;
978
979 return coro->subcoro == NULL || bco_reload_begin(coro->subcoro);
980}

982bool
983bco_reload_end(bco_t* coro) {
984 if (coro->status != BCO_SUSPENDED || coro->resume_point == 0) { return true; }
985 BCO_ASSERT(coro->relocating, "bco_reload_end was called without bco_reload_begin");
986
987 bco__relocate_call(coro);
988 coro->relocating = false;
989 if (!coro->relocate_found) {
990 coro->resume_point = BCO__RELOCATE; // Anything but 0 so bco_terminate runs the cleanup
991 bco_terminate(coro);
992 return false;
993 }
