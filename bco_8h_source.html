561bco_status_t
562bco_resume(bco_t* coro) {
563 if (coro->status != BCO_SUSPENDED) { return coro->status; }
564
565 coro->status = BCO_RUNNING;
566 coro->sp = coro->bp;
567 coro->fn(coro, coro->args);
568 BCO_ASSERT(coro->resume_point != 0, "Bare return was used to return from coroutine");
569 return coro->status = coro->resume_point >= 0 ? BCO_SUSPENDED : BCO_TERMINATED;
570}

583bco_terminate(bco_t* coro) {
584 if (coro->status != BCO_SUSPENDED) { return; }
585 if (coro->resume_point == 0) {

596 bco_resume(coro);
