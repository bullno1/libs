1162static inline bent_comp_save_t
1163bent_comp_save_mode(const bent_comp_def_t* def) {
1164 int raw = (def->flags & BENT_COMP_RAW) != 0;
1165 int transient = (def->flags & BENT_COMP_TRANSIENT) != 0;
1166 int callback = def->serialize != NULL;
1167 if (raw + transient + callback > 1) { return BENT_COMP_SAVE_INVALID; }
1168 if (transient) { return BENT_COMP_SAVE_NONE; }
1169 if (callback) { return BENT_COMP_SAVE_CALLBACK; }
1170 if (raw) { return BENT_COMP_SAVE_RAW; }
1171 if (def->size == 0) { return BENT_COMP_SAVE_PRESENCE; }
1172 return BENT_COMP_SAVE_INVALID;
1173}

1533static void
1534bent_sys_init(
1535 bent_world_t* world,
1536 bent_system_data_t* sys,
1537 const char* name, const bent_sys_def_t* def
1538) {
1539 sys->def = def;
1540
1541 BENT_LOG("Initializing %s", name);
1542
1543#ifndef BENT_NO_RELOAD
1544 // Backup old filter to detect change
1545 bent_bitset_t old_require = sys->require;
1546 bent_bitset_t old_exclude = sys->exclude;
1547#endif
1548
1549 if (sys->name == NULL) {
1550 sys->name = bent_strcpy(name, world->memctx);
1551 }
1552
1553 bent_bitset_clear(&sys->require);
1554 bent_bitset_clear(&sys->exclude);
1555
1556 if (def->require == NULL && def->exclude == NULL) {
1557 // A system that specifies nothing will not match anything
1558 bent_bitset_flip(&sys->require);
1559 bent_bitset_flip(&sys->exclude);
1560 } else {
1561 // Otherwise, each property is defaulted to an empty list
1562 sys->require = bent_bitset_from_comp_list(def->require);
1563 sys->exclude = bent_bitset_from_comp_list(def->exclude);
1564 }
1565
1566 bool initialized = sys->initialized;
1567 if (def->size) {
1568 sys->userdata = BENT_REALLOC(sys->userdata, def->size, world->memctx);
1569 if (!initialized) {
1570 memset(sys->userdata, 0, def->size);
1571 }
1572 }
1573
1574 if (def->init && (!initialized || (def->flags & BENT_SYS_ALLOW_REINIT))) {
1575 def->init(sys->userdata, world);
1576 }
1577
1578#ifndef BENT_NO_RELOAD
1579 // Update component inclusion based on new filter
1580 // Also, allow newly created systems to immediately register existing entities
1581 bent_system_data_t old_sys = {
1582 .require = old_require,
1583 .exclude = old_exclude,
1584 };
1585 bool outermost = bent_begin_notify(world);
1586 BHANDLE_FOREACH(handle, &world->handles) {
1587 const bent_entity_data_t* entity = bseg_ref(world->entities, handle.index);
1588 const bent_bitset_t* components = &entity->components;
1589 bent_t entity_id = bent__from_bhandle(handle);
1590
1591 if (sys->initialized) { // Existing system, do diff
1592 if (bent_sys_match_impl(&old_sys, components)) {
1593 if (!bent_sys_match_impl(sys, components)) {
1594 bent_sys_remove_entity(world, sys, entity_id);
1595 }
1596 } else {
1597 if (bent_sys_match_impl(sys, components)) {
1598 bent_sys_add_entity(world, sys, entity_id);
1599 }
1600 }
1601 } else { // Newly registered system, try to match existing entities
1602 if (bent_sys_match_impl(sys, components)) {
1603 bent_sys_add_entity(world, sys, entity_id);
1604 }
1605 }
1606 }
1607 if (outermost) { bent_end_notify(world); }
1608#endif
1609
1610 sys->initialized = true;
1611}

1789 &&
1790 comp->def->cleanup

1799bool
1800bent_init(bent_world_t** world_ptr, void* memctx) {
1801 bent_world_t* world = *world_ptr;
1802 bool first_init = world == NULL;
1803 if (world == NULL) {
1804 world = BENT_REALLOC(NULL, sizeof(bent_world_t), memctx);
1805 *world = (bent_world_t){
1806 .memctx = memctx,
1807 };
1808 }
1809
1810 AUTOLIST_FOREACH(itr, bent__components) {
1811 bent_comp_reg_t* reg = itr->value_addr;
1812#ifndef BENT_NO_RELOAD
1813 if (reg->id == 0) { // Unregistered or we just reloaded
1814 // Search existing components for a match by name
1815 bent_index_t num_components = world->num_components;
1816 for (bent_index_t i = 0; i < num_components; ++i) {
1817 const bent_component_data_t* comp = &world->components[i];
1818 if (strcmp(comp->name, itr->name) == 0) {
1819 reg->id = i + 1;
1820 break;
1821 }
1822 }
1823 }
1824#endif
1825
1826 // Still not found, register for the first time
1827 if (reg->id == 0) {
1828 reg->id = ++world->num_components; // 1-based
1829 BENT_ASSERT(world->num_components <= BENT_MAX_NUM_COMPONENT_TYPES);
1830 }
1831
1832 world->num_components = reg->id > world->num_components ? reg->id : world->num_components;
1833 bent_comp_init(
1834 world,
1835 &world->components[reg->id - 1],
1836 itr->name, reg->def
1837 );
1838 }
1839
1840 // Count the number of systems first
1841 bent_index_t num_systems = (bent_index_t)barray_len(world->systems);
1842 AUTOLIST_FOREACH(itr, bent__systems) {
1843 bent_sys_reg_t* reg = itr->value_addr;
1844#ifndef BENT_NO_RELOAD
1845 if (reg->id == 0) { // Unregistered or we just reloaded
1846 // Search existing systems for a match by name
1847 for (bent_index_t i = 0; i < (bent_index_t)barray_len(world->systems); ++i) {
1848 const bent_system_data_t* sys = &world->systems[i];
1849 if (strcmp(sys->name, itr->name) == 0) {
1850 reg->id = i + 1;
1851 break;
1852 }
1853 }
1854 }
1855#endif
1856
1857 // Still not found, register for the first time
1858 if (reg->id == 0) {
1859 reg->id = ++num_systems;
1860 }
1861
1862 num_systems = num_systems > reg->id ? num_systems : reg->id;
1863 }
1864
1865 // Do a single alloc
1866 if (num_systems > (bent_index_t)barray_len(world->systems)) {
1867 barray_resize(world->systems, num_systems, world->memctx);
1868 }
1869
1870 // (Re)initialize systems
1871 AUTOLIST_FOREACH(itr, bent__systems) {
1872 bent_sys_reg_t* reg = itr->value_addr;
1873
1874 bent_sys_init(
1875 world,
1876 &world->systems[reg->id - 1],
1877 itr->name, reg->def
1878 );
1879 }
1880
1881 // Post initialization
1882 AUTOLIST_FOREACH(itr, bent__systems) {
1883 bent_sys_reg_t* reg = itr->value_addr;
1884
1885 bent_system_data_t* sys = &world->systems[reg->id - 1];
1886 if (sys->def->post_init) {
1887 sys->def->post_init(sys->userdata, world);
1888 }
1889 }
1890
1891 *world_ptr = world;
1892 return first_init;
1893}

1896bent_cleanup(bent_world_t** world_ptr) {
1897 bent_world_t* world = *world_ptr;
1898 if (world == NULL) { return; }
1899
1900#ifndef BENT_NO_RELOAD
1901 // Edge case: cleaning up right after a reload
1902 bent_init(world_ptr, world->memctx);
1903 world = *world_ptr;
1904#endif
1905
1906 bent_clear(world);
1907
1908 bent_index_t num_components = world->num_components;
1909 for (bent_index_t i = 0; i < num_components; ++i) {
1910 bent_comp_cleanup(world, &world->components[i]);
1911 }

1914 for (bent_index_t i = 0; i < num_systems; ++i) {
1915 bent_sys_cleanup(world, &world->systems[i]);
1916 }

1919 barray_free(world->notify_queue, world->memctx);
1920 bseg_free(world->entities, world->memctx);
1921 bhandle_free(&world->handles, world->memctx);
1922 barray_free(world->destroy_queue, world->memctx);
1923 BENT_REALLOC(world, 0, world->memctx);
1924
1925 *world_ptr = NULL;
1926}

2002 }
2003

2005 bent_bitset_set(&entity_data->components, comp_index);
2006 bent_bitset_t new_components = entity_data->components;
2007 bent_notify_systems(world, entity_id, &old_components, &new_components);
2008
2009 return instance;
2010 } else {
2011 // Already added, return existing data
2012 return bent_comp_instance(comp_data, entity_id.index);
2013 }
