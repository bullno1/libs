1616static inline bent_comp_save_t
1617bent_comp_save_mode(const bent_comp_def_t* def) {
1618 int raw = (def->flags & BENT_COMP_RAW) != 0;
1619 int transient = (def->flags & BENT_COMP_TRANSIENT) != 0;
1620 int callback = def->serialize != NULL;
1621 if (raw + transient + callback > 1) { return BENT_COMP_SAVE_INVALID; }
1622 if (transient) { return BENT_COMP_SAVE_NONE; }
1623 if (callback) { return BENT_COMP_SAVE_CALLBACK; }
1624 if (raw) { return BENT_COMP_SAVE_RAW; }
1625 if (def->size == 0) { return BENT_COMP_SAVE_PRESENCE; }
1626 return BENT_COMP_SAVE_INVALID;
1627}
