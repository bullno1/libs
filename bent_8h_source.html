2497

2503// Idempotent: a query created in the middle of a load or a callback may
2504// already hold an entity that is notified later.
2505static void
2506bent_query_insert(bent_world_t* world, bent_query_data_t* query, bent_t entity) {
2507 if (bent_query_contains(query, entity)) { return; }
2508
2509 bent_index_t sparse_size = (bent_index_t)barray_len(query->sparse);
2510 if (entity.index >= sparse_size) {
2511 bent_index_t new_sparse_size = sparse_size * 2 > entity.index + 1 ? sparse_size * 2 : entity.index + 1;
2512 barray_resize(query->sparse, new_sparse_size, world->memctx);
2513 }
2514 query->sparse[entity.index] = (bent_index_t)barray_len(query->dense);
2515 barray_push(query->dense, entity, world->memctx);
2516}

2518static void
2519bent_query_erase(bent_query_data_t* query, bent_t entity) {
2520 if (!bent_query_contains(query, entity)) { return; }
2521
2522 bent_index_t dense_index = query->sparse[entity.index];
2523 bent_t last_entity = barray_pop(query->dense);
2524 if (dense_index < (bent_index_t)barray_len(query->dense)) {
2525 query->dense[dense_index] = last_entity;
2526 query->sparse[last_entity.index] = dense_index;
2527 }
2528}

2532 barray_free(query->dense, world->memctx);

2837static void

2839 // Callbacks may queue more while draining.
2840 // Each round drains what the previous one queued, into the spare pair so
2841 // that a payload being delivered is never moved by a reallocation.
2842 while (barray_len(world->notify_queue) > 0) {
2843 barray(bent_notification_t) queue = world->notify_queue;
2844 barray(char) payloads = world->msg_payloads;
2845 world->notify_queue = world->spare_notify_queue;
2846 world->msg_payloads = world->spare_msg_payloads;

2849 for (bent_index_t i = 0; i < len; ++i) {
