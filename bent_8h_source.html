1745static inline bent_comp_save_t
1746bent_comp_save_mode(const bent_comp_def_t* def) {
1747 int raw = (def->flags & BENT_COMP_RAW) != 0;
1748 int transient = (def->flags & BENT_COMP_TRANSIENT) != 0;
1749 int callback = def->serialize != NULL;
1750 if (raw + transient + callback > 1) { return BENT_COMP_SAVE_INVALID; }
1751 if (transient) { return BENT_COMP_SAVE_NONE; }

2440// Deliver whatever the callbacks queued, then the entities they destroyed
2441static void
2442bent_end_notify(bent_world_t* world) {
2443 // Callbacks may queue more while draining.
2444 // Each round drains what the previous one queued, into the spare pair so
2445 // that a payload being delivered is never moved by a reallocation.
2446 while (barray_len(world->notify_queue) > 0) {
2447 barray(bent_notification_t) queue = world->notify_queue;
2448 barray(char) payloads = world->msg_payloads;
2449 world->notify_queue = world->spare_notify_queue;
2450 world->msg_payloads = world->spare_msg_payloads;
2451
2452 bent_index_t len = (bent_index_t)barray_len(queue);
2453 for (bent_index_t i = 0; i < len; ++i) {
2454 bent_dispatch_notification(world, &queue[i], payloads);
2455 }
2456
2457 barray_clear(queue);
2458 barray_clear(payloads);
2459 world->spare_notify_queue = queue;
2460 world->spare_msg_payloads = payloads;
2461 }
2462 world->notifying = false;
2463
2464 if (!world->defer_destruction) {
2465 bent_drain_destroy_queue(world);
2466 }
