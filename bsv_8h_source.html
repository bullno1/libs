455static inline bsv_status_t
456bsv_read(bsv_in_t* in, void* buf, size_t size) {
457 char* cbuf = buf;
458 while (size > 0) {
459 size_t bytes_read = in->read(in, cbuf, size);
460 if (bytes_read == 0) { return BSV_IO_ERROR; }
461 cbuf += bytes_read;
462 size -= bytes_read;
463 }
464
465 return BSV_OK;
466}

469static inline bsv_status_t
470bsv_write(bsv_out_t* out, const void* buf, size_t size) {
471 const char* cbuf = buf;
472 while (size > 0) {
473 size_t bytes_written = out->write(out, cbuf, size);
474 if (bytes_written == 0) { return BSV_IO_ERROR; }
475 cbuf += bytes_written;
476 size -= bytes_written;
477 }
478
479 return BSV_OK;

776bsv_status_t
777bsv_write_uint(uint64_t x, bsv_out_t* out) {
778 char buf[10];
779 size_t n = 0;
780
781 for (int i = 0; i < 10; ++i) {
782 n += x >= 0x80;
783 buf[i] = (char)(x | 0x80);
784 x >>= 7;
785 }
786
787 buf[n] ^= 0x80;
788 n += 1;
789
790 return bsv_write(out, buf, n);
791}

794bsv_write_sint(int64_t x, bsv_out_t* out) {
795 uint64_t ux = ((uint64_t)x << 1) ^ (uint64_t)(x >> 63); // zigzag encode
796 return bsv_write_uint(ux, out);

799bsv_status_t
800bsv_read_uint(uint64_t* x, bsv_in_t* in) {
801 uint64_t b;
802 char c;
803 uint64_t tmp = 0;
804
805 for (int i = 0; i < 10; ++i) {
806 BSV_CHECK_STATUS(bsv_read(in, &c, 1));
807
808 b = c;
809 tmp |= (b & 0x7f) << (7 * i);
810 if (b < 0x80) {
811 *x = tmp;
812 return BSV_OK;
813 }
814 }
815
816 return BSV_MALFORMED;
817}
