Detailed Description

A command line option.

Example:

barg_opt_t opts[] = {
{
.name = "optimize",
.short_name = 'O',
.value_name = "level",
.summary = "Optimization level",
.parser = barg_int(&level),
},
{
.name = "verbose",
.short_name = 'v',
.summary = "Print more details",
.boolean = true,
.parser = barg_boolean(&verbose),
},
{
.name = "output",
.short_name = 'o',
.value_name = "file",
.summary = "Where to write the output",
.parser = barg_str(&output),
},
{
.name = "define",
.short_name = 'D',
.value_name = "name",
.summary = "Define a macro",
.repeatable = true,
.parser = barg_array(&defines_opts),
},
{
.name = "color",
.value_name = "mode",
.summary = "When to use colors",
.description =
"One of: auto, always, never.\n"
"Defaults to auto.",
.parser = {
.userdata = &color,
.parse = parse_color,
},
},
// The standard --help option
barg_opt_help(),
};
