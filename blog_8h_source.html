414void
415blog_init(const blog_options_t* options) {
416 blog_state.options = *options;
417
418 int depth = options->current_depth_in_project + 1;
419 const char* current_filename = options->current_filename;
420 if (current_filename != NULL) {
421 int len = (int)strlen(current_filename);
422 int i = len - 1;
423 for (; i >= 0; --i) {
424 char ch = current_filename[i];
425 if (ch == '/' || ch == '\\') {
426 --depth;
427 }
428
429 if (depth == 0) { break; }
430 }
431
432 if (
433 i >= 0
434 &&
435 (current_filename[i] == '/' || current_filename[i] == '\\')
436 ) {
437 i += 1;
438 }
439
440 if (depth == 0 && i >= 0) {
441 blog_state.prefix_len = i;
442 }
443 }
444}

483void
484blog_write_raw(
485 blog_level_t level,
486 const char* filename,
487 int line,
488 const char* msg,
489 int len
490) {
491 filename = filename != NULL ? filename : "<unknown>";
492 int filename_len = (int)strlen(filename);
493 blog_str_t filename_str = {
494 .len = filename_len,
495 .data = filename,
496 };
497
498 // Shorten filename by common prefix
499 if (
500 blog_state.prefix_len > 0
501 && filename_len >= blog_state.prefix_len
502 && memcmp(filename, blog_state.options.current_filename, blog_state.prefix_len) == 0
503 ) {
504 filename_str.len -= blog_state.prefix_len;
505 filename_str.data += blog_state.prefix_len;
506 }
507
508 blog_ctx_t ctx = {
509 .file = filename_str,
510 .line = line,
511 .level = level,
512 };
513 blog_str_t msg_str = {
514 .len = len,
515 .data = msg,
516 };
517
518 for (int i = 0; i < blog_state.num_loggers; ++i) {
519 blog_logger_t* logger = &blog_state.loggers[i];
520 if (level >= logger->min_level) {
521 logger->fn(&ctx, msg_str, logger->userdata);
522 }
523 }

531 const char* fmt,
532 va_list args
533) {
534 if (!blog_is_enabled(level)) { return; }
535

537 int msg_len = vsnprintf(line_buf, sizeof(line_buf), fmt, args);
538 if (msg_len < 0) {
539 msg_len = 0;
