353void
354blog_init(const blog_options_t* options) {
355 blog_state.options = *options;
356
357 int depth = options->current_depth_in_project + 1;
358 const char* current_filename = options->current_filename;
359 if (current_filename != NULL) {
360 int len = (int)strlen(current_filename);
361 int i = len - 1;
362 for (; i >= 0; --i) {
363 char ch = current_filename[i];
364 if (ch == '/' || ch == '\\') {
365 --depth;
366 }
367
368 if (depth == 0) { break; }
369 }
370
371 if (
372 i >= 0
373 &&
374 (current_filename[i] == '/' || current_filename[i] == '\\')
375 ) {
376 i += 1;
377 }
378
379 if (depth == 0 && i >= 0) {
380 blog_state.prefix_len = i;
381 }
382 }

412void
413blog_vwrite(
414 blog_level_t level,
415 const char* filename,
416 int line,
417 const char* fmt,
418 va_list args
419) {
420 filename = filename != NULL ? filename : "<unknown>";
421 int filename_len = (int)strlen(filename);
422 blog_str_t filename_str = {
423 .len = filename_len,
424 .data = filename,
425 };
426
427 // Shorten filename by common prefix
428 if (
429 blog_state.prefix_len > 0
430 && filename_len >= blog_state.prefix_len
431 && memcmp(filename, blog_state.options.current_filename, blog_state.prefix_len) == 0
432 ) {
433 filename_str.len -= blog_state.prefix_len;
434 filename_str.data += blog_state.prefix_len;
435 }
436
437 blog_ctx_t ctx = {
438 .file = filename_str,
439 .line = line,
440 .level = level,
441 };
442
443 int msg_len = -1;
444 for (int i = 0; i < blog_state.num_loggers; ++i) {
445 blog_logger_t* logger = &blog_state.loggers[i];
446 if (level >= logger->min_level) {
447 // Delay formatting until it's actually needed
448 if (msg_len < 0) {
449 msg_len = vsnprintf(
450 blog_state.line_buf, sizeof(blog_state.line_buf),
451 fmt, args
452 );
453 if (msg_len < 0) {
454 msg_len = 0;
455 } else if (msg_len >= (int)sizeof(blog_state.line_buf)) {
456 msg_len = sizeof(blog_state.line_buf) - 1;
457 }
458 blog_state.line_buf[msg_len] = '\0';
459 }
460
461 blog_str_t msg = {
462 .len = msg_len,
463 .data = blog_state.line_buf
464 };
465 logger->fn(&ctx, msg, logger->userdata);
466 }
467 }
