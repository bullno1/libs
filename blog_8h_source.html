391void
392blog_init(const blog_options_t* options) {
393 blog_state.options = *options;
394
395 int depth = options->current_depth_in_project + 1;
396 const char* current_filename = options->current_filename;
397 if (current_filename != NULL) {
398 int len = (int)strlen(current_filename);
399 int i = len - 1;
400 for (; i >= 0; --i) {
401 char ch = current_filename[i];
402 if (ch == '/' || ch == '\\') {
403 --depth;
404 }
405
406 if (depth == 0) { break; }
407 }
408
409 if (
410 i >= 0
411 &&
412 (current_filename[i] == '/' || current_filename[i] == '\\')
413 ) {
414 i += 1;
415 }
416
417 if (depth == 0 && i >= 0) {
418 blog_state.prefix_len = i;
419 }
420 }

450void
451blog_vwrite(
452 blog_level_t level,
453 const char* filename,
454 int line,
455 const char* fmt,
456 va_list args
457) {
458 filename = filename != NULL ? filename : "<unknown>";
459 int filename_len = (int)strlen(filename);
460 blog_str_t filename_str = {
461 .len = filename_len,
462 .data = filename,
463 };
464
465 // Shorten filename by common prefix
466 if (
467 blog_state.prefix_len > 0
468 && filename_len >= blog_state.prefix_len
469 && memcmp(filename, blog_state.options.current_filename, blog_state.prefix_len) == 0
470 ) {
471 filename_str.len -= blog_state.prefix_len;
472 filename_str.data += blog_state.prefix_len;
473 }
474
475 blog_ctx_t ctx = {
476 .file = filename_str,
477 .line = line,
478 .level = level,
479 };
480
481 int msg_len = -1;
482 for (int i = 0; i < blog_state.num_loggers; ++i) {
483 blog_logger_t* logger = &blog_state.loggers[i];
484 if (level >= logger->min_level) {
485 // Delay formatting until it's actually needed
486 if (msg_len < 0) {
487 msg_len = vsnprintf(
488 blog_line_buf, sizeof(blog_line_buf),
489 fmt, args
490 );
491 if (msg_len < 0) {
492 msg_len = 0;
493 } else if (msg_len >= (int)sizeof(blog_line_buf)) {
494 msg_len = sizeof(blog_line_buf) - 1;
495 }
496 blog_line_buf[msg_len] = '\0';
497 }
498
499 blog_str_t msg = {
500 .len = msg_len,
501 .data = blog_line_buf
502 };
503 logger->fn(&ctx, msg, logger->userdata);
504 }
505 }
506}
